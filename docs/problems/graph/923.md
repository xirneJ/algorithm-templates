# AcWing 854. Floyd求最短路

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
#define N 210
#define MAX 0x3f3f3f3f
int d[N][N]={0};
int main()
{
    int i,j,k,n,m,q;
    cin>>n>>m>>q;
    for(i=1;i<=n;i++)
    {
        for(j=1;j<=n;j++)
        {
            if(i==j)
            {
                d[i][j]=0;
            }
            else
            {
                d[i][j]=MAX;
            }
        }
    }
    while(m--)
    {
        int l,r,v;
        cin>>l>>r>>v;
        d[l][r]=min(d[l][r],v);
    }
    for(k=1;k<=n;k++)
    {
        for(i=1;i<=n;i++)
        {
            for(j=1;j<=n;j++)
            {
                d[i][j]=min(d[i][j],d[i][k]+d[k][j]);
            }
        }
    }
    while(q--)
    {
        int l,r;
        cin>>l>>r;
        if(d[l][r]>MAX/2)
        {
            printf("impossible\n");
        }
        else
        {
            cout<<d[l][r]<<endl;
        }
    }
    return 0;
}
```
