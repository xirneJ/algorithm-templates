# AcWing 861. 二分图的最大匹配

## 模板

```cpp
#include <iostream>
#include <algorithm>
#include <cstring>
using namespace std;
#define N 510
#define M 200010
int idx=0;
int ne[M]={0};
int np[M]={0};
int h[N]={0};
int st[N]={0};
int match[N]={0};
void add(int a,int b)
{
    np[idx]=b;
    ne[idx]=h[a];
    h[a]=idx++;
}
int find(int x)
{
    int i;
    for(i=h[x];i!=-1;i=ne[i])
    {
        int r=np[i];
        if(!st[r])
        {
            st[r]=1;
            if(match[r]==0||find(match[r]))
            {
                match[r]=x;
                return 1;
            }
        }
    }
    return 0;
}
int main()
{
    memset(h,-1,sizeof(h));
    int n1,n2,m;
    cin>>n1>>n2>>m;
    int i;
    for(i=0;i<m;i++)
    {
        int l,r;
        cin>>l>>r;
        add(l,r);
    }
    int res=0;
    for(i=1;i<=n1;i++)
    {
        memset(st,0,sizeof(st));
        if(find(i))
        {
            res++;
        }
    }
    cout<<res<<endl;
    return 0;
}
```
