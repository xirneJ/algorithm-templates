# AcWing 848. 有向图的拓扑序列

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#define N 100010
int ne[N]={0};
int np[N]={0};
int h[N]={0};
int idx=0;
void add(int a,int b)
{
    np[idx]=b;
    ne[idx]=h[a];
    h[a]=idx++;
}
int team[N]={0};
int sym[N]={0};
int count[N]={0};
int n=0;
int m=0;
int left=0;
int right=-1;
int tp()
{
    int i=0;
    for(i=1;i<=n;i++)
    {
        if(count[i]==0)
        {
            team[++right]=i;
        }
    }
    while(left<=right)
    {
        int l=team[left];
        for(i=h[l];i!=-1;i=ne[i])
        {
            int r=np[i];
            count[r]--;
            if(count[r]==0)
            {
                team[++right]=r;
            }
        }
        left++;
    }
    if(right==n-1)
    {
        return 1;
    }
    else
    {
        return -1;
    }
}
int main()
{
    scanf("%d %d",&n,&m);
    memset(h,-1,sizeof(int)*(n+1));
    int i=0;
    for(i=0;i<m;i++)
    {
        int l=0;
        int r=0;
        scanf("%d %d",&l,&r);
        add(l,r);
        count[r]++;
    }
    int res=tp();
    if(res==1)
    {
        for(i=0;i<=right;i++)
    {
        printf("%d ",team[i]);
    }
    }
    else
    {
        printf("-1");
    }
    return 0;
}
```
