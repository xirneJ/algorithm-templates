# AcWing 849. Dijkstra求最短路 I

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#define N 510
#define M 100010
#define MAX 0x3f3f3f3f
int ne[M]={0};
int np[M]={0};
int h[N]={0};
int idx=0;
int dis[N]={0};
int sym[N]={0};
int valu[M]={0};
int n=0;
int m=0;
void add(int a,int b,int c)
{
    np[idx]=b;
    ne[idx]=h[a];
    valu[idx]=c;
    h[a]=idx++;
}
int min(int a,int b)
{
    return a<b?a:b;
}
int dj()
{
    int i=0;
    int j=0;
    for(i=0;i<n;i++)
    {
        int p=-1;
         for(j=1;j<=n;j++)
         {
             if(!sym[j]&&(p==-1||dis[j]<dis[p]))
             {
                p=j;
             }
         }
        sym[p]=1;
        for(j=h[p];j!=-1;j=ne[j])
        {
            int r=np[j];
            if(sym[r]==0)
            {
                dis[r]=min(dis[r],dis[p]+valu[j]);
            }
        }
    }
    if(dis[n]==MAX)
    {
        return 0;
    }
    else
    {
        return 1;
    }
}
int main()
{
    scanf("%d %d",&n,&m);
    int i=0;
    memset(h,-1,sizeof(int)*(n+1));
    for(i=0;i<m;i++)
    {
        int l=0;
        int r=0;
        int v=0;
        scanf("%d %d %d",&l,&r,&v);
        add(l,r,v);
    }
    for(i=2;i<=n;i++)
    {
        dis[i]=MAX;
    }
    dis[1]=0;
    int res=dj();
    if(res==1)
    {
        printf("%d",dis[n]);
    }
    else
    {
        printf("-1");
    }
    return 0;
}
```
