# AcWing 852. spfa判断负环

## 模板

```cpp
#include <iostream>
#include <algorithm>
#include <cstring>
#include <queue>
using namespace std;
#define MAX 0x3f3f3f3f
#define N 2010
#define M 10010
int h[N];
int ne[M];
int np[M];
int idx=0;
int w[M];
int st[N];
int cnt[N];
int dis[N];
void add(int a,int b,int c)
{
    np[idx]=b;
    w[idx]=c;
    ne[idx]=h[a];
    h[a]=idx++;
}
int n,m;
int spfa()
{
    memset(dis,MAX,sizeof(dis));
    int i;
    dis[1]=0;
    queue<int>s;
    for(i=1;i<=n;i++)
    {
        s.push(i);
        st[i]=1;
    }
    while(s.size())
    {
        int t=s.front();
        s.pop();
        st[t]=0;
        for(i=h[t];i!=-1;i=ne[i])
        {
            int r=np[i];
            if(dis[r]>dis[t]+w[i])
            {
                dis[r]=dis[t]+w[i];
                cnt[r]=cnt[t]+1;
                if(cnt[r]>=n)
                {
                    return 1;
                }
                if(!st[r])
                {
                    st[r]=1;
                    s.push(r);
                }
            }
        }
    }
    return 0;
}
int main()
{
    cin>>n>>m;
    memset(h,-1,sizeof(h));
    while(m--)
    {
        int a,b,c;
        cin>>a>>b>>c;
        add(a,b,c);
    }
    if(spfa())
    {
        cout<<"Yes"<<endl;
    }
    else
    {
        cout<<"No"<<endl;
    }
    return 0;
}
```
