# AcWing 787. 归并排序

## 模板

```cpp
#include <iostream>
#include <bits/stdc++.h>
#include <cstdlib>
using namespace std;
#define N 100010
void gb(int*arr,int left,int right)
{
    if(left>=right)
    {
        return;
    }
    int mid=(right-left)/2+left;
    gb(arr,left,mid);
    gb(arr,mid+1,right);
    int*temp=(int*)malloc(sizeof(int)*(right-left+1));
    int k=0;
    int i=left;
    int j=mid+1;
    while(i<=mid&&j<=right)
    {
        if(arr[i]<arr[j])
        {
            temp[k++]=arr[i++];
        }
        else
        {
            temp[k++]=arr[j++];
        }
    }
    while(i<=mid)
    {
        temp[k++]=arr[i++];
    }
    while(j<=right)
    {
        temp[k++]=arr[j++];
    }
    for(i=0;i<k;i++)
    {
        arr[left+i]=temp[i];
    }
    free(temp);
}
int main()
{
    int n=0;
    scanf("%d",&n);
    int i=0;
    int arr[N]={0};
    for(i=0;i<n;i++)
    {
        scanf("%d",&arr[i]);
    }
    gb(arr,0,n-1);
    for(i=0;i<n;i++)
    {
        printf("%d ",arr[i]);
    }
    return 0;
}
```
