# AcWing 785. 快速排序

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#define N 100010
void quick_sort(int*arr,int left,int right)
{
    if(left>=right)
    {
        return;
    }
    int x=arr[(right-left)/2+left];
    int i=left-1;
    int j=right+1;
    while(i<j)
    {
        do
        {
            i++;
        }while(arr[i]<x);
        do
        {
            j--;
        }while(arr[j]>x);
        if(i<j)
        {
            int num=arr[i];
            arr[i]=arr[j];
            arr[j]=num;
        }
    }
    quick_sort(arr,left,j);
    quick_sort(arr,j+1,right);
}
int main()
{
    int n=0;
    int arr[N]={0};
    scanf("%d",&n);
    int i=0;
    for(i=0;i<n;i++)
    {
        scanf("%d",&arr[i]);
    }
    quick_sort(arr,0,n-1);
    for(i=0;i<n;i++)
    {
        printf("%d ",arr[i]);
    }
    return 0;
}
```
