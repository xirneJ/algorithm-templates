# AcWing 786. 第k个数

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#define N 100010
int find(int*arr,int left,int right,int k)
{
    if(left==right)
    {
        return arr[left];
    }
    int x=arr[(right-left)/2+left];
    int i=left-1;
    int j=right+1;
    while(i<j)
    {
        do
        {
            i++;
        }while(arr[i]<x);
        do
        {
            j--;
        }while(arr[j]>x);
        if(i<j)
        {
            int num=arr[i];
            arr[i]=arr[j];
            arr[j]=num;
        }
    }
    if(j<k)
    {
        return find(arr,j+1,right,k);
    }
    else
    {
        return find(arr,left,j,k);
    }
}
int main()
{
    int n=0;
    int k=0;
    scanf("%d %d",&n,&k);
    int i=0;
    int arr[N]={0};
    k--;
    for(i=0;i<n;i++)
    {
        scanf("%d",&arr[i]);
    }
    int res=find(arr,0,n-1,k);
    printf("%d",res);
    return 0;
}
```
