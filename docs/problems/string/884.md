# AcWing 143. 最大异或对

## 模板

```cpp
#include <iostream>
#include <string>
#include <algorithm>
using namespace std;
#define N 100010
#define M 3001000
int arr[N]={0};
int son[M][2]={0};
int idx=0;
void insert(int x)
{
    int p=0;
    int i;
    for(i=30;i>=0;i--)
    {
        if(!son[p][x>>i&1])
        {
            son[p][x>>i&1]=++idx;
        }
        p=son[p][x>>i&1];
    }
}
int query(int x)
{
    int p=0;
    int i;
    int res=0;
    for(i=30;i>=0;i--)
    {
        if(son[p][!(x>>i&1)])
        {
            res=res+(1<<i);
            p=son[p][!(x>>i&1)];
        }
        else
        {
            p=son[p][x>>i&1];
        }
    }
    return res;
}
int main()
{
    int n,i;
    cin>>n;
    for(i=0;i<n;i++)
    {
        cin>>arr[i];
        insert(arr[i]);
    }
    int res=0;
    for(i=0;i<n;i++)
    {
        res=max(res,query(arr[i]));
    }
    cout<<res<<endl;
    return 0;
}
```
