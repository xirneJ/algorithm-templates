# AcWing 835. Trie字符串统计

## 模板

```cpp
#include <iostream>
#include <string>
using namespace std;
#define N 100010
int son[N][26]={0};
int cnt[N]={0};
int idx=0;
void insert(string str)
{
    int p=0;
    int i;
    for(i=0;str[i];i++)
    {
        int u=str[i]-'a';
        if(!son[p][u])
        {
            son[p][u]=++idx;
        }
        p=son[p][u];
    }
    cnt[p]++;
}
int query(string str)
{
    int p=0;
    int i;
    for(i=0;str[i];i++)
    {
        int u=str[i]-'a';
        if(!son[p][u])
        {
            return 0;
        }
        p=son[p][u];
    }
    return cnt[p];
}
int main()
{
    int t;
    cin>>t;
    while(t--)
    {
        string str;
        string str1;
        cin>>str>>str1;
        if(str.compare("I")==0)
        {
            insert(str1);
        }
        else if(str.compare("Q")==0)
        {
            int res=query(str1);
            cout<<res<<endl;
        }
    }
    return 0;
}
```
