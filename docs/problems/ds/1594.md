# AcWing 243. 一个简单的整数问题2

## 模板

```cpp
#include <algorithm>
#include <iostream>
using namespace std;
#define N 100010
typedef long long ll;
int a[N]={0};
int n,m;
ll tree1[N]={0};//维护差分的前缀和
ll tree2[N]={0};//维护差分*i的前缀和
int lowbit(int x)
{
    return x&-x;
}
ll sum(ll*tree,int x)
{
    int i;
    ll res=0;
    for(i=x;i;i-=lowbit(i))
    {
        res+=tree[i];
    }
    return res;
}
void add(ll*tree,int x,ll d)
{
    int i;
    for(i=x;i<=n;i+=lowbit(i))
    {
        tree[i]+=d;
    }
}
ll pre_sum(int x)
{
    return sum(tree1,x)*(ll)(x+1)-sum(tree2,x);
}
int main()
{
    cin>>n>>m;
    int i;
    for(i=1;i<=n;i++)
    {
        cin>>a[i];
    }
    for(i=1;i<=n;i++)
    {
        int b=a[i]-a[i-1];
        add(tree1,i,b);
        add(tree2,i,(ll)b*i);
    }
    while(m--)
    {
        string str;
        cin>>str;
        if(str[0]=='Q')
        {
            int l,r;
            cin>>l>>r;
            cout<<pre_sum(r)-pre_sum(l-1)<<endl;
        }
        else
        {
            int l,r,d;
            cin>>l>>r>>d;
            add(tree1,l,(ll)d);
            add(tree2,l,(ll)d*l);
            if(r+1<=n)
            {
                add(tree1,r+1,-d);
                add(tree2,r+1,(-d)*(r+1));
            }
        }
    }
    return 0;
}
```
