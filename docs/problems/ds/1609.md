# AcWing 246. 区间最大公约数

## 模板

```cpp
#include <iostream>
#include <algorithm>
#include <string>
#include <cmath>
using namespace std;
#define N 500010
typedef long long ll;
struct node
{
    int l,r;
    ll g;
    ll sum;
}tree[4*N];
ll w[N]={0};
ll gcd(ll a,ll b)
{
    return b?gcd(b,a%b):a;
}
void push_up(node&root,node&left,node&right)
{
    root.g=gcd(left.g,right.g);
    root.sum=left.sum+right.sum;
}
void pushup(int u)
{
    push_up(tree[u],tree[u*2],tree[u*2+1]);
}
void build(int u,int l,int r)
{
    if(l==r)
    {
        tree[u]={l,r,w[l],w[l]};
        return;
    }
    else
    {
        tree[u].l=l;
        tree[u].r=r;
        int mid=l+r>>1;
        build(u*2,l,mid);
        build(u*2+1,mid+1,r);
        pushup(u);
    }
}
void modify(int u,int x,ll v)
{
    if(tree[u].l==x&&tree[u].r==x)
    {
        ll b=tree[u].sum+v;
        tree[u]={x,x,b,b};
    }
    else
    {
        int mid=tree[u].l+tree[u].r>>1;
        if(x<=mid)
        {
            modify(u*2,x,v);
        }
        else
        {
            modify(u*2+1,x,v);
        }
        pushup(u);
    }
}
node query(int u,int l,int r)
{
    if(tree[u].l>=l&&tree[u].r<=r)
    {
        return tree[u];
    }
    else
    {
        int mid=tree[u].l+tree[u].r>>1;
        if(r<=mid)
        {
            return query(u*2,l,r);
        }
        else if(l>mid)
        {
            return query(u*2+1,l,r);
        }
        else
        {
            node left=query(u*2,l,r);
            node right=query(u*2+1,l,r);
            node res;
            push_up(res,left,right);
            return res;
        }
    }
}
int main()
{
    int n,m;
    cin>>n>>m;
    int i;
    ll last=0;
    ll now=0;
    for(i=1;i<=n;i++)
    {
        cin>>now;
        w[i]=now-last;
        last=now;
    }
    build(1,1,n);
    while(m--)
    {
        string str;
        int l,r;
        cin>>str>>l>>r;
        if(str[0]=='Q')
        {
            node left=query(1,1,l);
            node right={0};
            if(l+1<=r)
            {
                right=query(1,l+1,r);
            }
            cout<<abs(gcd(left.sum,right.g))<<endl;
        }
        else if(str[0]=='C')
        {
            ll d;
            cin>>d;
            modify(1,l,d);
            if(r+1<=n)
            {
                modify(1,r+1,-d);
            }
        }
    }
    return 0;
}
```
