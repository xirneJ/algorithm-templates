# AcWing 245. 你能回答这些问题吗

## 模板

```cpp
#include <iostream>
#include <algorithm>
#include <string>
using namespace std;
#define N 500010
struct node
{
    int l,r;
    int sum,lsum,rsum,tmax;
}tree[4*N];
int w[N]={0};
void push_up(node& root,node& left,node& right)
{
    root.sum=left.sum+right.sum;
    root.lsum=max(left.lsum,left.sum+right.lsum);
    root.rsum=max(right.rsum,right.sum+left.rsum);
    root.tmax=max(max(left.tmax,right.tmax),left.rsum+right.lsum);
}
void pushup(int u)
{
    push_up(tree[u],tree[u*2],tree[u*2+1]);
}
void build(int u,int l,int r)
{
    if(l==r)
    {
        tree[u]={l,l,w[l],w[l],w[l],w[l]};
        return;
    }
    else
    {
        tree[u].l=l;
        tree[u].r=r;
        int mid=(l+r)/2;
        build(u*2,l,mid);
        build(u*2+1,mid+1,r);
        pushup(u);
    }
}
void modify(int u,int x,int v)
{
    if(tree[u].l==x&&tree[u].r==x)
    {
        tree[u]={x,x,v,v,v,v};
    }
    else
    {
        int mid=(tree[u].l+tree[u].r)/2;
        if(x<=mid)
        {
            modify(u*2,x,v);
        }
        else if(x>mid)
        {
            modify(u*2+1,x,v);
        }
        pushup(u);
    }
}
node query(int u,int l,int r)
{
    if(tree[u].l>=l&&tree[u].r<=r)
    {
        return tree[u];
    }
    else
    {
        int mid=(tree[u].l+tree[u].r)/2;
        if(r<=mid)
        {
            return query(u*2,l,r);
        }
        else if(l>mid)
        {
            return query(u*2+1,l,r);
        }
        else
        {
            node left=query(u*2,l,r);
            node right=query(u*2+1,l,r);
            node res;
            push_up(res,left,right);
            return res;
        }
    }
}
int main()
{
    int n,m;
    cin>>n>>m;
    int i;
    for(i=1;i<=n;i++)
    {
        cin>>w[i];
    }
    build(1,1,n);
    while(m--)
    {
        int k,x,y;
        cin>>k>>x>>y;
        if(k==1)
        {
            if(x>y)
            {
                swap(x,y);
            }
            cout<<query(1,x,y).tmax<<endl;
        }
        else
        {
            modify(1,x,y);
        }
    }
    return 0;
}
```
