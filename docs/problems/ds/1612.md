# AcWing 1277. 维护序列

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
typedef long long ll;
#define N 100010
struct node
{
    int l,r;
    ll sum,add,mul;
}tree[N*4];
ll p;
ll w[N];
void pushup(int u)
{
    tree[u].sum=(tree[u*2].sum+tree[u*2+1].sum)%p;
}
void push_down(node&son,ll add,ll mul)
{
    son.sum=(son.sum*mul%p+(son.r-son.l+1)*add%p)%p;//取模的优先级比较高，要用括号
    son.add=(son.add*mul%p+add%p)%p;//区分乘法和加法的运算优先级
    son.mul=son.mul*mul%p;
}
void pushdown(int u)
{
    push_down(tree[u*2],tree[u].add,tree[u].mul);
    push_down(tree[u*2+1],tree[u].add,tree[u].mul);
    tree[u].add=0;
    tree[u].mul=1;
}
void build(int u,int l,int r)
{
    if(l==r)
    {
        tree[u]={l,r,w[l],0,1};
    }
    else
    {
        tree[u]={l,r,0,0,1};
        int mid=l+r>>1;
        build(u*2,l,mid);
        build(u*2+1,mid+1,r);
        pushup(u);
    }
}
void modify_add(int u,int l,int r,ll add)
{
    if(tree[u].l>=l&&tree[u].r<=r)
    {
        tree[u].sum=(tree[u].sum+(tree[u].r-tree[u].l+1)*add%p)%p;
        tree[u].add=(tree[u].add+add%p)%p;
    }
    else
    {
        pushdown(u);
        int mid=tree[u].l+tree[u].r>>1;
        if(mid>=l)
        {
            modify_add(u*2,l,r,add);
        }
        if(mid<r)
        {
            modify_add(u*2+1,l,r,add);
        }
        pushup(u);
    }
}
void modify_mul(int u,int l,int r,ll mul)
{
    if(tree[u].l>=l&&tree[u].r<=r)
    {
        tree[u].sum=tree[u].sum*mul%p;
        tree[u].mul=tree[u].mul*mul%p;
        tree[u].add=tree[u].add*mul%p;//乘法时要要更行加是应为运算优先级
    }
    else
    {
        pushdown(u);
        int mid=tree[u].l+tree[u].r>>1;
        if(mid>=l)
        {
            modify_mul(u*2,l,r,mul);
        }
        if(mid<r)
        {
            modify_mul(u*2+1,l,r,mul);
        }
        pushup(u);
    }
}
ll query(int u,int l,int r)
{
    if(tree[u].l>=l&&tree[u].r<=r)
    {
        return tree[u].sum;
    }
    else
    {
        pushdown(u);
        int mid=tree[u].l+tree[u].r>>1;
        ll res=0;
        if(mid>=l)
        {
            res=(res+query(u*2,l,r))%p;
        }
        if(mid<r)
        {
            res=(res+query(u*2+1,l,r))%p;
        }
        return res;
    }
}
int main()
{
    ll n;
    cin>>n>>p;
    int i;
    for(i=1;i<=n;i++)
    {
        scanf("%lld",&w[i]);
    }
    ll m;
    cin>>m;
    build(1,1,n);
    while(m--)
    {
        int op;
        cin>>op;
        ll t,g,c;
        if(op==1)
        {
            cin>>t>>g>>c;
            modify_mul(1,t,g,c);
        }
        else if(op==2)
        {
            cin>>t>>g>>c;
            modify_add(1,t,g,c);
        }
        else if(op==3)
        {
            cin>>t>>g;
            ll res=query(1,t,g);
            cout<<res<<endl;
        }
    }
    return 0;
}
```
