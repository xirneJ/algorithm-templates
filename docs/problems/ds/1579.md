# AcWing 1250. 格子游戏

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
#define N 500010
int p[N]={0};
int find(int x)
{
    if(p[x]==x)
    {
        return x;
    }
    else
    {
        return p[x]=find(p[x]);
    }
}
int main()
{
    int i;
    int n,m;
    cin>>n>>m;
    for(i=1;i<=(n*n+n);i++)
    {
        p[i]=i;
    }
    int cnt=1;
    while(m--)
    {
        int x,y;
        cin>>x>>y;
        int a,b,pa,pb;
        a=x*n+y;
        pa=find(a);
        char str[3];
        cin>>str;
        if(str[0]=='D')
        {
            b=a+n;
            pb=find(b);
            if(pa==pb)
            {
                cout<<cnt<<endl;
                return 0;
            }
            else
            {
                p[pa]=pb;
                cnt++;
            }
        }
        else if(str[0]=='R')
        {
            b=a+1;
            pb=find(b);
            if(pa==pb)
            {
                cout<<cnt<<endl;
                return 0;
            }
            else
            {
                p[pa]=pb;
                cnt++;
            }
        }
    }
    cout<<"draw"<<endl;
    return 0;
}
```
