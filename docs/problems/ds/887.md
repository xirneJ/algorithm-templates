# AcWing 240. 食物链

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
#define N 50010
int p[N]={0};
int dis[N]={0};
int find(int x)
{
    if(p[x]!=x)
    {
        int u=find(p[x]);
        dis[x]+=dis[p[x]];
        p[x]=u;
    }
    return p[x];
}
int main()
{
    int n,q;
    cin>>n>>q;
    int res=0;
    int i;
    for(i=1;i<=n;i++)
    {
        p[i]=i;
    }
    while(q--)
    {
        int d,x,y;
        cin>>d>>x>>y;
        if(x>n||y>n)
        {
            res++;
            continue;
        }
        if(d==1)
        {
            int px=find(x);
            int py=find(y);
            if(px==py)
            {
                if((dis[x]-dis[y])%3)
                {
                    res++;
                }
            }
            else
            {
                p[px]=py;
                dis[px]=dis[y]-dis[x];
            }
        }
        else if(d==2)
        {
            int px=find(x);
            int py=find(y);
            if(px==py)
            {
                if((dis[x]-dis[y]-1)%3)
                {
                    res++;
                }
            }
            else
            {
                p[px]=py;
                dis[px]=dis[y]-dis[x]+1;
            }
        }
    }
    cout<<res<<endl;
    return 0;
}
```
