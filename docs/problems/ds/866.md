# AcWing 829. 模拟队列

## 模板

```cpp
#include <bits/stdc++.h>
#include <iostream>
using namespace std;
int main()
{
    int n=0;
    queue <int>s;
    cin>>n;
    while(n--)
    {
        char arr[10]={0};
        scanf("%s",arr);
        if(strcmp(arr,"push")==0)
        {
            int num=0;
            cin>>num;
            s.push(num);
        }
        else if(strcmp(arr,"pop")==0)
        {
            s.pop();
        }
        else if(strcmp(arr,"empty")==0)
        {
            if(s.empty())
            {
                cout<<"YES"<<endl;
            }
            else
            {
                cout<<"NO"<<endl;
            }
        }
        else if(strcmp(arr,"query")==0)
        {
            int num=s.front();
            cout<<num<<endl;
        }
    }
    return 0;
}
```
