# AcWing 830. 单调栈

## 模板

```cpp
#include <iostream>
#include <algorithm>
#include <stack>
using namespace std;
#define N 100010
int arr[N]={0};
int main()
{
    stack <int>s;
    int n,i;
    cin>>n;
    for(i=0;i<n;i++)
    {
        scanf("%d",&arr[i]);
    }
    for(i=0;i<n;i++)
    {
        while(!s.empty()&&arr[i]<=s.top())
        {
            s.pop();
        }
        if(s.empty())
        {
            printf("-1 ");
        }
        else
        {
            printf("%d ",s.top());
        }
        s.push(arr[i]);
    }
    return 0;
}
```
