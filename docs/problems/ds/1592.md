# AcWing 241. 楼兰图腾

## 模板

```cpp
#include <iostream>
#include <cstring>
#include <algorithm>
using namespace std;
#define N 200010
typedef long long ll;
int a[N]={0};
int tree[N]={0};//记录<=i的树出现的次数
int Greater[N]={0};//记录这个数前面比这个数大的数
int Lower[N]={0};//记录这个数前面比这个数小的数
int n;
int lowbit(int x)
{
    return x&(-x);
}
void add(int x,int d)
{
    int i;
    for(i=x;i<=n;i+=lowbit(i))
    {
        tree[i]+=d;
    }
}
int sum(int x)
{
    int i;
    ll res=0;
    for(i=x;i;i-=lowbit(i))
    {
        res+=tree[i];
    }
    return res;
}
int main()
{
    cin>>n;
    int i;
    for(i=1;i<=n;i++)
    {
        cin>>a[i];
    }
    for(i=1;i<=n;i++)
    {
        int y=a[i];
        Greater[i]=sum(n)-sum(y);
        Lower[i]=sum(y-1);
        add(y,1);
    }
    ll res1=0;
    ll res2=0;
    memset(tree,0,sizeof(tree));
    for(i=n;i>=1;i--)
    {
        int y=a[i];
        res1=res1+Greater[i]*(ll)(sum(n)-sum(y));
        res2=res2+Lower[i]*(ll)sum(y-1);
        add(y,1);
    }
    printf("%lld %lld",res1,res2);
    return 0;
}
```
