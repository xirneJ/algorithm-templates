# AcWing 836. 合并集合

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
#define N 100010
int p[N]={0};
int find(int x)
{
    if(p[x]==x)
    {
        return x;
    }
    return p[x]=find(p[x]);
}
int main()
{
    int n,m;
    int i=0;
    cin>>n>>m;
    for(i=1;i<=n;i++)
    {
        p[i]=i;
    }
    while(m--)
    {
        char a[10]={0};
        cin>>a;
        if(a[0]=='M')
        {
            int a,b;
            cin>>a>>b;
            int pa=find(a);//注意这一段，不能p[a]=b，也不能pa=p[a],pb=p[b],p[pa]=pb;
            int pb=find(b);
            p[pa]=pb;
        }
        else if(a[0]=='Q')
        {
            int a,b;
            cin>>a>>b;
            int pa=find(a);
            int pb=find(b);
            if(pa==pb)
            {
                printf("Yes\n");
            }
            else
            {
                printf("No\n");
            }
        }
    }
    return 0;
}
```
