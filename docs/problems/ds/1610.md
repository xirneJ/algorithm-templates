# AcWing 243. 一个简单的整数问题2

## 模板

```cpp
#include <iostream>
#include <algorithm>
#include <cstdio>
#include <string>
using namespace std;
typedef long long ll;
#define N 100010
struct node
{
    int l,r;
    ll sum,add;
}tree[4*N];
ll w[N];
void pushup(int u)
{
    tree[u].sum=tree[u*2].sum+tree[u*2+1].sum;
}
void pushdown(int u)
{
    node&root=tree[u];
    node&left=tree[u*2];
    node&right=tree[u*2+1];
    if(root.add)//root.add可能为负数
    {
        ll d=root.add;
        left.add+=d;
        left.sum=left.sum+(left.r-left.l+1)*d;
        right.add+=d;
        right.sum=right.sum+(right.r-right.l+1)*d;
        root.add=0;
    }
}
void build(int u,int l,int r)
{
    if(l==r)
    {
        tree[u]={l,r,w[l],0};
    }
    else
    {
        tree[u].l=l;
        tree[u].r=r;
        int mid=l+r>>1;
        build(u*2,l,mid);
        build(u*2+1,mid+1,r);
        pushup(u);
    }
}
void modify(int u,int l,int r,ll d)
{
    if(tree[u].l>=l&&tree[u].r<=r)
    {
        tree[u].sum+=(ll)(tree[u].r-tree[u].l+1)*d;
        tree[u].add+=d;
    }
    else
    {
        pushdown(u);
        int mid=tree[u].l+tree[u].r>>1;
        if(mid<r)//两种可能都有可能进入
        {
            modify(u*2+1,l,r,d);
        }
        if(mid>=l)
        {
            modify(u*2,l,r,d);
        }
        pushup(u);
    }
}
ll query(int u,int l,int r)
{
    if(tree[u].l>=l&&tree[u].r<=r)
    {
        return tree[u].sum;
    }
    else
    {
        pushdown(u);
        int mid=tree[u].l+tree[u].r>>1;
        ll sum=0;
        if(mid>=l)
        {
            sum+=query(u*2,l,r);//两种可能都有可能进入
        }
        if(mid<r)
        {
            sum+=query(u*2+1,l,r);
        }
        return sum;
    }
}
int main()
{
    int n,m;
    cin>>n>>m;
    int i;
    for(i=1;i<=n;i++)
    {
        scanf("%lld",&w[i]);
    }
    build(1,1,n);
    while(m--)
    {
        char str[10]={0};
        int l,r;
        cin>>str>>l>>r;
        if(str[0]=='Q')
        {
            printf("%lld\n",query(1,l,r));
        }
        else if(str[0]=='C')
        {
            ll d;
            cin>>d;
            modify(1,l,r,d);
        }
    }
    return 0;
}
```
