# AcWing 840. 模拟散列表

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
typedef struct l
{
    struct l*next_node;
    int v;
}arry;
#define N 100010
arry*arr[N]={0};
int find(int p,int x)
{
    arry*now_node=arr[p];
    while(now_node!=NULL)
    {
        if(now_node->v==x)
        {
            return 1;
        }
        now_node=now_node->next_node;
    }
    return 0;
}
arry*get(int v)
{
    arry*new_node=malloc(sizeof(arry));
    new_node->next_node=NULL;
    new_node->v=v;
    return new_node;
}
void insert(int p,int x)
{
    if(arr[p]==NULL)
    {
        arr[p]=get(x);
    }
    else
    {
        arry*now_node=arr[p];
        while(now_node->next_node!=NULL)
        {
            now_node=now_node->next_node;
        }
        now_node->next_node=get(x);
    }
}
int main()
{
    int n=0;
    scanf("%d",&n);
    while(n--)
    {
        char str[10]={0};
        scanf("%s",str);
        if(strcmp(str,"I")==0)
        {
            int x,p;
            scanf("%d",&x);
            p=(x+1000000000)%100000;
            insert(p,x);

        }
        else if(strcmp(str,"Q")==0)
        {
            int x,p;
            scanf("%d",&x);
            p=(x+1000000000)%100000;
            if(find(p,x))
            {
                printf("Yes\n");
            }
            else
            {
                printf("No\n");
            }
        }
    }
    return 0;
}
```
