# AcWing 1252. 搭配购买

## 模板

```cpp
#include <iostream>
#include <algorithm>
#include <string>
#include <cstring>
using namespace std;
#define N 10010
int cost[N];
int valu[N];
int p[N];
int sum;
int dp[N];
int find(int x)
{
    if(p[x]==x)
    {
        return x;
    }
    else
    {
        return p[x]=find(p[x]);
    }
}
int main()
{
    int i,j;
    int n,m;
    cin>>n>>m>>sum;
    for(i=1;i<=n;i++)
    {
        p[i]=i;
    }
    for(i=1;i<=n;i++)
    {
        cin>>cost[i];
        cin>>valu[i];
    }
    while(m--)
    {
        int a,b;
        cin>>a>>b;
        int pa=find(a);
        int pb=find(b);
        if(pa!=pb)
        {
            p[pa]=pb;
            cost[pb]+=cost[pa];
            valu[pb]+=valu[pa];
        }
    }
    for(i=1;i<=n;i++)
    {
        if(p[i]==i)
        {
            for(j=sum;j>=cost[i];j--)
            {
                dp[j]=max(dp[j],dp[j-cost[i]]+valu[i]);
            }
        }
    }
    cout<<dp[sum]<<endl;
    return 0;
}
```
