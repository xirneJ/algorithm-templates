# AcWing 828. 模拟栈

## 模板

```cpp
#include <iostream>
#include <bits/stdc++.h>
using namespace std;
int main()
{
    int n=0;
    stack <int>s;
    cin>>n;
    while(n--)
    {
        char arr[10]={0};
        scanf("%s",arr);
        if(strcmp(arr,"push")==0)
        {
            int num=0;
            cin>>num;
            s.push(num);
        }
        else if(strcmp(arr,"pop")==0)
        {
            s.pop();
        }
        else if(strcmp(arr,"empty")==0)
        {
            int num=s.empty();
            if(num==1)
            {
                cout<<"YES"<<endl;
            }
            else
            {
                cout<<"NO"<<endl;
            }
        }
        else if(strcmp(arr,"query")==0)
        {
            int num=s.top();
            cout<<num<<endl;
        }
    }
    return 0;
}
```
