# AcWing 154. 滑动窗口

## 模板

```cpp
#include <iostream>//t1找最小值，t2找最大值
#include <algorithm>
#include <queue>
using namespace std;
#define N 1000010
int arr[N]={0};
int t1[N]={0};
int t2[N]={0};
int main()
{
    int n,k;
    cin>>n>>k;
    int l1=0;
    int l2=0;
    int r1=-1;
    int r2=-1;
    int i=0;
    for(i=0;i<n;i++)
    {
        cin>>arr[i];
    }
    for(i=0;i<n;i++)
    {
        if(i-t1[l1]>=k)
        {
            l1++;
        }
        while(l1<=r1&&arr[i]<=arr[t1[r1]])
        {
            r1--;
        }
        t1[++r1]=i;
        if(i>=k-1)
        {
            printf("%d ",arr[t1[l1]]);
        }
    }
    printf("\n");
    for(i=0;i<n;i++)
    {
        if(i-t2[l2]>=k)
        {
            l2++;
        }
        while(l2<=r2&&arr[i]>=arr[t2[r2]])
        {
            r2--;
        }
        t2[++r2]=i;
        if(i>=k-1)
        {
            printf("%d ",arr[t2[l2]]);
        }
    }
    return 0;
}
```
