# AcWing 237. 程序自动分析

## 模板

```cpp
#include <iostream>
#include <algorithm>
#include <string>
#include <cstring>
#include <unordered_map>
using namespace std;
#define N 200010
int p[N];
int find(int x)
{
    if(p[x]==x)
    {
        return x;
    }
    else
    {
        return p[x]=find(p[x]);
    }
}
int len=1;
unordered_map<int,int>s;
struct node
{
    int x,y,e;
};
int get(int x)
{
    if(s.count(x)==0)
    {
        s[x]=++len;
    }
    return s[x];
}
int main()
{
    int t;
    cin>>t;
    int i;
    while(t--)
    {
        len=0;//每次要重置len
        node arr[200010]={0};
        s.clear();
        int n;
        cin>>n;
        for(i=0;i<n;i++)
        {
            int x,y,e;
            cin>>x>>y>>e;
            arr[i]={get(x),get(y),e};
        }
        for(i=1;i<=len;i++)//注意这里len是刚好指向最后一个元素还是指向最后一个元素的后一个元素
        {
            p[i]=i;
        }
        for(i=0;i<n;i++)
        {
            if(arr[i].e==1)
            {
                int px=find(arr[i].x);
                int py=find(arr[i].y);
                if(px!=py)
                {
                    p[px]=py;
                }
            }
        }
        int flag=0;
        for(i=0;i<n;i++)
        {
            if(arr[i].e==0)
            {
                int px=find(arr[i].x);
                int py=find(arr[i].y);
                if(px==py)
                {
                    flag=1;
                    break;
                }
            }
        }
        if(flag)
        {
            cout<<"NO"<<endl;
        }
        else
        {
            cout<<"YES"<<endl;
        }
    }
    return 0;
}
```
