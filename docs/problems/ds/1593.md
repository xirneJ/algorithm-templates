# AcWing 242. 一个简单的整数问题

## 模板

```cpp
#include <iostream>
#include <algorithm>
#include <string>
typedef long long ll;
using namespace std;
#define N 100010
ll tree[N]={0};
ll a[N]={0};
int n;
int lowbit(int x)
{
    return x&(-x);
}
void add(int x,int c)
{
    int i;
    for(i=x;i<=n;i+=lowbit(i))
    {
        tree[i]+=(ll)c;
    }
}
ll sum(int x)
{
    ll res=0;
    int i;
    for(i=x;i;i-=lowbit(i))
    {
        res+=(ll)tree[i];
    }
    return res;
}
int main()
{
    int m;
    cin>>n>>m;
    int i;
    int now=0;
    int last=0;
    for(i=1;i<=n;i++)
    {
        cin>>now;
        a[i]=now-last;
        last=now;
        add(i,a[i]);
    }
    while(m--)
    {
        string str;
        cin>>str;
        if(str[0]=='Q')
        {
            int x;
            cin>>x;
            cout<<sum(x)<<endl;
        }
        else
        {
            int l,r,d;
            cin>>l>>r>>d;
            add(l,d);
            if(r+1<=n)
            {
                add(r+1,-d);
            }
        }
    }
    return 0;
}
```
