# AcWing 838. 堆排序

## 模板

```cpp
#include <iostream>
#include <queue>
#include <algorithm>
using namespace std;
int cmp(int a,int b)
{
    return a>b;
}
int main()
{
    int n,m;
    cin>>n>>m;
    priority_queue<int,vector<int>,greater<int>>s;
    int i=0;
    for(i=0;i<n;i++)
    {
        int a;
        cin>>a;
        s.push(a);
    }
    while(m--)
    {
        cout<<s.top()<<" ";
        s.pop();
    }
    return 0;
}
```
