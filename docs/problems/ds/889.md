# AcWing 839. 模拟堆

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#define N 100010
typedef struct l
{
    int p;
    int v;
}arry;
arry arr[N]={0};
int len=0;
int min(int a,int b)
{
    return a<b?a:b;
}
void down_move(int x);
void up_move(int x)
{
    int p=x/2;
    if(p>=1&&arr[x].v<arr[p].v)
    {
        arry a=arr[x];
        arr[x]=arr[p];
        arr[p]=a;
        up_move(p);
        down_move(x);
    }
}
void down_move(int x)
{
    int l=x*2;
    int r=x*2+1;
    int m=x;
    if(l<=len&&arr[l].v<arr[m].v)
    {
        m=l;
    }
    if(r<=len&&arr[r].v<arr[m].v)
    {
        m=r;
    }
    if(m!=x)
    {
        arry a=arr[m];
        arr[m]=arr[x];
        arr[x]=a;
        down_move(m);
        up_move(x);
    }
}
int idx=1;
int main()
{
    int n;
    int i;
    scanf("%d",&n);
    while(n--)
    {
        char str[10]={0};
        scanf("%s",str);
        if(strcmp(str,"I")==0)
        {
            int x;
            scanf("%d",&x);
            len++;
            arr[len].v=x;
            arr[len].p=idx++;
            up_move(len);//应该传len而非arr[len].p前者是位置，后者是编号
        }
        else if(strcmp(str,"PM")==0)
        {
            printf("%d\n",arr[1].v);
        }
        else if(strcmp(str,"DM")==0)
        {
            arr[1]=arr[len];
            len--;
            down_move(1);
        }
        else if(strcmp(str,"D")==0)//注意元素个数要减少
        {
            int pos;
            scanf("%d",&pos);
            for(i=1;i<=len;i++)
            {
                if(arr[i].p==pos)
                {
                    if(arr[len].v>arr[i].v)
                    {
                        arr[i]=arr[len];
                        len--;
                        down_move(i);
                        break;
                    }
                    else if(arr[len].v<arr[i].v)
                    {
                        arr[i]=arr[len];
                        len--;
                        up_move(i);
                        break;
                    }
                    else if(arr[len].v==arr[i].v)
                    {
                        arr[i]=arr[len];
                        len--;
                        break;
                    }
                }
            }
        }
        else if(strcmp(str,"C")==0)//注意元素个数要减少
        {
            int pos,x;
            scanf("%d %d",&pos,&x);
            for(i=1;i<=len;i++)
            {
                if(arr[i].p==pos)
                {
                    if(x<arr[i].v)
                    {
                        arr[i].v=x;
                        up_move(i);
                        break;
                    }
                    else if(x>arr[i].v)
                    {
                        arr[i].v=x;
                        down_move(i);
                        break;
                    }
                }
            }
        }
    }
    return 0;
}
```
