# AcWing 238. 银河英雄传说

## 模板

```cpp
#include <iostream>
#include <algorithm>
#include <cmath>
using namespace std;
#define N 30010
int p[N];
int s[N];
int d[N];
int find(int x)
{
    if(p[x]!=x)
    {
        int root=find(p[x]);
        d[x]+=d[p[x]];
        p[x]=root;
    }
    return p[x];
}
int main()
{
    int m,i;
    cin>>m;
    for(i=1;i<=N;i++)
    {
        p[i]=i;
        s[i]=1;
    }
    while(m--)
    {
        string str;
        int a,b;
        cin>>str>>a>>b;
        if(str[0]=='M')
        {
            int pa=find(a);
            int pb=find(b);
            if(pa!=pb)//这里要判断一下是不是在同一个集合里面
            {
                d[pa]=s[pb];//这里是直接赋值，不需要加
                s[pb]+=s[pa];
                p[pa]=pb;
            }
        }
        else
        {
            int pa=find(a);
            int pb=find(b);
            if(pa!=pb)
            {
                cout<<"-1"<<endl;
            }
            else
            {
                cout<<max(0,abs(d[a]-d[b])-1)<<endl;//简化成这样
            }
        }
    }
    return 0;
}
```
