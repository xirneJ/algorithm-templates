# AcWing 239. 奇偶游戏

## 模板

```cpp
#include <iostream>
#include <algorithm>
#include <unordered_map>
using namespace std;
#define N 100010
int p[N];
int d[N];
int find(int x)
{
    if(p[x]!=x)
    {
        int root=find(p[x]);
        d[x]^=d[p[x]];
        p[x]=root;
    }
    return p[x];
}
unordered_map<int,int>s;
int len=0;
int get(int x)
{
    if(s.count(x)==0)
    {
        s[x]=++len;
    }
    return s[x];
}
int main()
{
    int n,m;
    cin>>n>>m;
    int i;
    for(i=1;i<=100000;i++)
    {
        p[i]=i;
    }
    int res=m;
    for(i=1;i<=m;i++)
    {
        int a,b;
        string str;
        cin>>a>>b>>str;
        a=get(a-1);
        b=get(b);
        int t=0;
        if(str=="odd")
        {
            t=1;
        }
        int pa=find(a);
        int pb=find(b);
        if(pa==pb)
        {
            if(d[a]^d[b]!=t)
            {
                res=i-1;
                break;
            }
        }
        else
        {
            p[pa]=pb;
            d[pa]=d[a]^d[b]^t;
        }
    }
    cout<<res<<endl;
    return 0;
}
```
