# AcWing 837. 连通块中点的数量

## 模板

```cpp
#include <iostream>
#include <algorithm>
#include <cstdlib>
#include <cstring>
using namespace std;
#define N 100010
int p[N]={0};
int find(int x)
{
    if(p[x]==x)
    {
        return x;
    }
    return p[x]=find(p[x]);
}
int main()
{
    int count[N]={0};
    int n,m;
    cin>>n>>m;
    int i,j;
    for(i=1;i<=n;i++)
    {
        p[i]=i;
        count[i]=1;
    }
    while(m--)
    {
        char ch[10]={0};
        cin>>ch;
        if(ch[0]=='C')
        {
            int a,b;
            cin>>a>>b;
            if(a!=b)
            {
                int pa=find(a);
                int pb=find(b);
                if(pa!=pb)//不在一个连通块中再合并
                {
                    p[pa]=pb;
                    count[pb]+=count[pa];
                }
            }
        }
        else if(ch[1]=='1')
        {
            int a,b;
            cin>>a>>b;
            int pa=find(a);
            int pb=find(b);
            if(pa==pb)
            {
                printf("Yes\n");
            }
            else
            {
                printf("No\n");
            }
        }
        else if(ch[1]=='2')
        {
            int a;
            cin>>a;
            int pa=find(a);
            cout<<count[pa]<<endl;
        }
    }
    return 0;
}
```
