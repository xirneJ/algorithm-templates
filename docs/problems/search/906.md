# AcWing 843. n-皇后问题

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
int n;
int st[15][15];
int x_st[25];
int y_st[25];
int l_st[25];
int r_st[25];
void dps(int deep)
{
    int i,j;
    if(deep==n+1)
    {
        for(i=1;i<=n;i++)
        {
            for(j=1;j<=n;j++)
            {
                if(st[i][j])
                {
                    cout<<"Q";
                }
                else
                {
                    cout<<".";
                }
            }
            cout<<endl;
        }
        cout<<endl;
        return;
    }
    for(i=1;i<=n;i++)
    {
        if(!st[deep][i]&&!y_st[i]&&!l_st[deep+i]&&!r_st[deep-i+n])
        {
            st[deep][i]=1;
            y_st[i]=1;
            l_st[deep+i]=1;
            r_st[deep-i+n]=1;
            dps(deep+1);
            st[deep][i]=0;
            y_st[i]=0;
            l_st[deep+i]=0;
            r_st[deep-i+n]=0;
        }
    }
}
int main()
{
    cin>>n;
    dps(1);
    return 0;
}
```
