# AcWing 842. 排列数字

## 模板

```cpp
#include <iostream>
#include <stack>
using namespace std;
int arr[10]={0};
int st[10]={0};
int len=0;
void dps(int deep,int n)
{
    int i;
    if(deep==n)
    {
        for(i=0;i<deep;i++)
        {
            printf("%d ",arr[i]);
        }
        printf("\n");
        return;
    }
    for(i=1;i<=n;i++)
    {
        if(!st[i])
        {
            st[i]=1;
            arr[len++]=i;
            dps(deep+1,n);
            len--;
            st[i]=0;
        }
    }
}
int main()
{
    int n;
    cin>>n;
    dps(0,n);
    return 0;
}
```
