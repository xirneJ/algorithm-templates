# AcWing 165. 小猫爬山

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
#include <string>
#define N 30
int cat[N]={0};
int res=20;
int n,m;
int sum[N]={0};
void dps(int u,int k)
{
    int i;
    if(k>=res)
    {
        return;
    }
    else if(u==n)
    {
        res=k;
        return;
    }
    for(i=0;i<k;i++)
    {
        if(sum[i]+cat[u]<=m)
        {
            sum[i]+=cat[u];
            dps(u+1,k);
            sum[i]-=cat[u];
        }
    }
    sum[k]=cat[u];
    dps(u+1,k+1);
    sum[k]=0;
}
int main()
{
    cin>>n>>m;
    int i;
    for(i=0;i<n;i++)
    {
        cin>>cat[i];
    }
    sort(cat,cat+n);
    reverse(cat,cat+n);
    dps(0,0);
    cout<<res<<endl;
    return 0;
}
```
