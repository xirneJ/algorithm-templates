# AcWing 869. 试除法求约数

## 模板

```cpp
//这个是朴素算法的求约数算法
#include <stdio.h>
#include <stdlib.h>
int cmp(const void*e1,const void*e2)
{
    int*a1=(int*)e1;
    int*a2=(int*)e2;
    return (*a1)-(*a2);
}
int main()
{
    int n=0;
    int i=0;
    int j=0;
    scanf("%d",&n);
    for(i=0;i<n;i++)
    {
        int num=0;
        int len=0;
        scanf("%d",&num);
        int arr[10000]={0};
        for(j=1;j<=num/j;j++)
        {
            if(num%j==0)
            {
                arr[len++]=j;
                if(num/j!=j)
                {
                    arr[len++]=num/j;
                }
            }
        }
        qsort(arr,len,sizeof(int),cmp);
        for(j=0;j<len;j++)
        {
            printf("%d ",arr[j]);
        }
        printf("\n");
    }
    return 0;
}
```
