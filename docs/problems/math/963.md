# AcWing 893. 集合-Nim游戏

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#define N 10010
#define M 110
int f[N]={0};
int arr[M]={0};
int n=0;
int m=0;
int g(int x)
{
    if(f[x]!=-1)
    {
        return f[x];
    }
    int i=0;
    int st[N]={0};
    for(i=0;i<n;i++)
    {
        if(x>=arr[i])
        {
            int t=g(x-arr[i]);
            st[t]=1;
        }
    }
    for(i=0;i<10010;i++)
    {
        if(st[i]==0)
        {
            return f[x]=i;
        }
    }
    return 0;
}
int main()
{
    memset(f,-1,sizeof(f));
    int i=0;
    scanf("%d",&n);
    for(i=0;i<n;i++)
    {
        scanf("%d",&arr[i]);
    }
    int res=0;
    scanf("%d",&m);
    for(i=0;i<m;i++)
    {
        int num=0;
        scanf("%d",&num);
        res=res^g(num);
    }
    if(res==0)
    {
        printf("No\n");
    }
    else
    {
        printf("Yes\n");
    }
    return 0;
}
```
