# AcWing 876. 快速幂求逆元

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
typedef long long ll;
ll qmi(ll a,ll k,ll p)
{
    ll res=1;
    while(k)
    {
        if(k&1)
        {
            res=res*a%p;
        }
        k=k>>1;
        a=a*a%p;
    }
    return res;
}
int main()
{
    int n=0;
    scanf("%d",&n);
    while(n--)
    {
        ll a,p,res;
        scanf("%lld %lld",&a,&p);
        res=qmi(a,p-2,p);
        if(a%p==0)
        {
            printf("impossible\n");
        }
        else
        {
            printf("%lld\n",res);
        }
    }
    return 0;
}
```
