# AcWing 877. 扩展欧几里得算法

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
int mgcd(int a,int b,int*x,int*y)
{
    if(b==0)
    {
        (*x)=1;
        (*y)=0;
        return a;
    }
    else
    {
        int d=mgcd(b,a%b,y,x);
        (*y)=(*y)-(a/b*(*x));
        return d;
    }
}
int main()
{
    int n=0;
    scanf("%d",&n);
    while(n--)
    {
        int a,b;
        int x=0;
        int y=0;
        scanf("%d %d",&a,&b);
        mgcd(a,b,&x,&y);
        printf("%d %d\n",x,y);
    }
    return 0;
}
```
