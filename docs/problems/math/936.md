# AcWing 867. 分解质因数

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
void divp(int num)
{
    int i=0;
    for(i=2;i<=num/i;i++)
    {
        if(num%i==0)
        {
            int s=0;
            while(num%i==0)
            {
                s++;
                num/=i;
            }
            printf("%d %d\n",i,s);
        }
    }
    if(num>1)
    {
        printf("%d 1\n",num);
    }
    printf("\n");
}
int main()
{
    int n=0;
    scanf("%d",&n);
    int i=0;
    for(i=0;i<n;i++)
    {
        int num=0;
        scanf("%d",&num);
        divp(num);
    }
    return 0;
}
```
