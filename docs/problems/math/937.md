# AcWing 868. 筛质数

## 模板

```cpp
//埃氏筛
#include <stdio.h>
#include <stdlib.h>
#define N 1000010
int st[N]={0};
int main()
{
    int i=0;
    int j=0;
    int count=0;
    int n=0;
    scanf("%d",&n);
    for(i=2;i<=n;i++)
    {
        if(st[i]==0)
        {
            count++;
            for(j=i*2;j<=n;j+=i)
            {
                st[j]=1;
            }
        }
    }
    printf("%d\n",count);
    return 0;
}

//欧氏筛
#include <stdio.h>
#include <stdlib.h>
#define N 1000010
int st[N]={0};
int prim[N]={0};
int main()
{
    int i=0;
    int j=0;
    int n=0;
    int len=0;
    scanf("%d",&n);
    for(i=2;i<=n;i++)
    {
        if(!st[i])
        {
            prim[len++]=i;
        }
        for(j=0;prim[j]<=n/i;j++)
        {
            st[prim[j]*i]=1;
            if(i%prim[j]==0)
            {
                break;
            }
        }
    }
    printf("%d\n",len);
    return 0;
}
```
