# AcWing 874. 筛法求欧拉函数

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#define N 1000010
typedef long long ll;
int prim[N]={0};
int sym[N]={0};
ll phi[N]={0};
int n=0;
int len=0;
ll get()
{
    int i=0;
    int j=0;
    phi[1]=1;
    ll res=0;
    for(i=2;i<=n;i++)
    {
        if(!sym[i])
        {
            prim[len++]=i;
            phi[i]=i-1;
        }
        for(j=0;prim[j]<=n/i;j++)
        {
            sym[prim[j]*i]=1;
            if(i%prim[j]==0)
            {
                phi[i*prim[j]]=phi[i]*prim[j];
                break;
            }
            phi[i*prim[j]]=phi[i]*(prim[j]-1);
        }
    }
    for(i=1;i<=n;i++)
    {
        res+=phi[i];
    }
    return res;
}
int main()
{
    scanf("%d",&n);
    ll res=get();
    printf("%lld",res);
    return 0;
}
```
