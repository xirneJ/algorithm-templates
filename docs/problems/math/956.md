# AcWing 886. 求组合数 II

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
typedef long long ll;
ll f[100010]={0};
ll nf[100010]={0};
ll qmi(ll a,ll k,ll p)
{
    ll res=1;
    while(k)
    {
        if(k&1)
        {
            res=res*a%p;
        }
        k>>=1;
        a=a*a%p;
    }
    return res;
}
int main()
{
    f[0]=1;
    nf[0]=1;
    int i=0;
    for(i=1;i<=100000;i++)
    {
        f[i]=(f[i-1]*i)%1000000007;
        nf[i]=(nf[i-1]*qmi(i,1000000005,1000000007))%1000000007;
    }
    int n=0;
    scanf("%d",&n);
    while(n--)
    {
        int a,b;
        scanf("%d %d",&a,&b);
        printf("%lld\n",f[a]*nf[b]%1000000007*nf[a-b]%1000000007);
    }
    return 0;
}
```
