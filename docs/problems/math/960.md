# AcWing 890. 能被整除的数

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
typedef long long ll;
int main()
{
    int n,m;
    scanf("%d %d",&n,&m);
    int p[20]={0};
    int i=0;
    int j=0;
    for(i=0;i<m;i++)
    {
        scanf("%d",&p[i]);
    }
    ll res=0;
    for(i=1;i<1<<m;i++)
    {
        ll t=1;
        ll count=0;
        for(j=0;j<m;j++)
        {
            if(i>>j&1)
            {
                count++;
                if(t*p[j]>n)
                {
                    t=-1;
                    break;
                }
                t=t*p[j];
            }
        }
        if(t!=-1)
        {
            if(count%2==1)
            {
                res=res+n/t;
            }
            else if(count%2==0)
            {
                res=res-n/t;
            }
        }
    }
    printf("%lld\n",res);
    return 0;
}
```
