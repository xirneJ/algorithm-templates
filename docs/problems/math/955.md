# AcWing 885. 求组合数 I

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
int dp[2010][2010]={0};
int n=0;
void get()
{
    int i=0;
    int j=0;
    for(i=0;i<2010;i++)
    {
        for(j=0;j<=i;j++)
        {
            if(j==0)
            {
                dp[i][j]=1;
            }
            if(i==0)
            {
                dp[i][j]=1;
            }
            else
            {
                dp[i][j]=(dp[i-1][j-1]+dp[i-1][j])%1000000007;
            }
        }
    }
}
int main()
{
    scanf("%d",&n);
    get();
    while(n--)
    {
        int a=0;
        int b=0;
        scanf("%d %d",&a,&b);
        printf("%d\n",dp[a][b]);
    }
    return 0;
}
```
