# AcWing 870. 约数个数

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
typedef struct l
{
    int prim;
    int count;
}arry;
arry arr[10000]={0};
typedef long long ll;
int len=0;
void add(int num)
{
    int i=0;
    for(i=0;i<len;i++)
    {
        if(arr[i].prim==num)
        {
            arr[i].count++;
            return;
        }
    }
    len++;
    arr[len-1].prim=num;
    arr[len-1].count=1;
}
int main()
{
    int n=0;
    scanf("%d",&n);
    int i=0;
    int j=0;
    for(i=0;i<n;i++)
    {
        int num=0;
        scanf("%d",&num);
        for(j=2;j<=num/j;j++)
        {
            while(num%j==0)
            {
                add(j);
                num/=j;
            }
        }
        if(num>1)
        {
            add(num);
        }
    }
    ll res=1;
    for(i=0;i<len;i++)
    {
        res=(res*(arr[i].count+1))%1000000007;
    }
    printf("%lld",res);
    return 0;
}
```
