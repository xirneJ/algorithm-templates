# AcWing 894. 拆分-Nim游戏

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#define N 1100
int f[N]={0};
int n=0;
int arr[N]={0};
int g(int x)
{
    if(f[x]!=-1)
    {
        return f[x];
    }
    int i=0;
    int j=0;
    int st[N]={0};
    for(i=0;i<x;i++)
    {
        for(j=0;j<=i;j++)
        {
            int t=g(i)^g(j);
            st[t]=1;
        }
    }
    for(i=0;;i++)
    {
        if(!st[i])
        {
            return f[x]=i;
        }
    }
}
int main()
{
    memset(f,-1,sizeof(f));
    scanf("%d",&n);
    int i=0;
    for(i=0;i<n;i++)
    {
        scanf("%d",&arr[i]);
    }
    int res=0;
    for(i=0;i<n;i++)
    {
        res=res^g(arr[i]);
    }
    if(res==0)
    {
        printf("No\n");
    }
    else
    {
        printf("Yes\n");
    }
    return 0;
}
```
