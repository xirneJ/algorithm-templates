# AcWing 866. 试除法判定质数

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
int is_prime(int num)
{
    int i=0;
    if(num<2)
    {
        return 0;
    }
    for(i=2;i<=num/i;i++)
    {
        if(num%i==0)
        {
            return 0;
        }
    }
    return 1;
}
int main()
{
    int n=0;
    scanf("%d",&n);
    int i=0;
    for(i=0;i<n;i++)
    {
        int num=0;
        scanf("%d",&num);
        if(is_prime(num))
        {
            printf("Yes\n");
        }
        else
        {
            printf("No\n");
        }
    }
    return 0;
}
```
