# AcWing 887. 求组合数 III

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
typedef long long ll;
ll p;
ll qmi(ll a,ll k,ll p)
{
    ll res=1;
    while(k)
    {
        if(k&1)
        {
            res=res*a%p;
        }
        k>>=1;
        a=a*a%p;
    }
    return res;
}
ll c(ll a,ll b)
{
    ll res=1;
    int i=0;
    int j=0;
    for(i=a,j=1;j<=b;j++,i--)
    {
        res=res*i%p*qmi(j,p-2,p)%p;
    }
    return res;
}
ll locas(ll a,ll b)
{
    if(a<p&&b<p)
    {
        return c(a,b);
    }
    else
    {
        return c(a%p,b%p)*locas(a/p,b/p)%p;
    }
}
int main()
{
    ll a,b,res;
    int n=0;
    scanf("%d",&n);
    while(n--)
    {
        scanf("%lld %lld %lld",&a,&b,&p);
        res=locas(a,b);
        printf("%lld\n",res);
    }
    return 0;
}
```
