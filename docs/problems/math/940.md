# AcWing 871. 约数之和

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
typedef long long ll;
typedef struct l
{
    int prim;
    int count;
}arry;
arry arr[100000]={0};
int len=0;
void add(int num)
{
    int i=0;
    for(i=0;i<len;i++)
    {
        if(arr[i].prim==num)
        {
            arr[i].count++;
            return ;
        }
    }
    len++;
    arr[len-1].prim=num;
    arr[len-1].count=1;
}
int main()
{
    int n=0;
    scanf("%d",&n);
    int i=0;
    int j=0;
    int num=0;
    while(n--)
    {
        scanf("%d",&num);
        for(i=2;i<=num/i;i++)
        {
            while(num%i==0)
            {
                add(i);
                num/=i;
            }
        }
        if(num>1)
        {
            add(num);
        }
    }
    ll res=1;
    for(i=0;i<len;i++)
    {
        int p=arr[i].prim;
        int c=arr[i].count;
        ll t=1;
        for(j=0;j<c;j++)
        {
            t=(t*p+1)%1000000007;
        }
        res=res*t%1000000007;
    }
    printf("%lld",res);
    return 0;
}
```
