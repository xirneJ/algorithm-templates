# AcWing 889. 满足条件的01序列

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
typedef long long ll;
ll qmi(ll a,ll k,ll p)
{
    ll res=1;
    while(k)
    {
        if(k&1)
        {
            res=res*a%p;
        }
        k>>=1;
        a=a*a%p;
    }
    return res;
}
int main()
{
    int n=0;
    scanf("%d",&n);
    int i=0;
    int j=0;
    ll res=1;
    for(i=2*n,j=1;j<=n;j++,i--)
    {
        res=res*i%1000000007*qmi(j,1000000005,1000000007)%1000000007;
    }
    res=res*qmi(n+1,1000000005,1000000007)%1000000007;
    printf("%lld",res);
    return 0;
}
```
