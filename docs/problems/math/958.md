# AcWing 888. 求组合数 IV

## 模板

```cpp
#include <stdio.h>
#define N 5010
int prim[N]={0};
int count[N]={0};
int st[N]={0};
int len1=0;
int len2=0;
void get_prim(int n)
{
    int i=0;
    int j=0;
    for(i=2;i<=n;i++)
    {
        if(!st[i])
        {
            prim[len1++]=i;
        }
        for(j=0;prim[j]<=n/i;j++)
        {
            st[prim[j]*i]=1;
            if(i%prim[j]==0)
            {
                break;
            }
        }
    }
}
int get(int a,int p)
{
    int res=0;
    while(a)
    {
        res=res+a/p;
        a/=p;
    }
    return res;
}
void reverse(int*r)
{
    int i=0;
    for(i=0;i<len2/2;i++)
    {
        int num=r[i];
        r[i]=r[len2-1-i];
        r[len2-1-i]=num;
    }
}
void mul(int* r,int b)
{
    int t=0;
    int i=0;
    reverse(r);
    for(i=0;i<len2;i++)
    {
        t=t+r[i]*b;
        r[i]=t%10;
        t=t/10;
    }
    while(t)
    {
        r[len2++]=t%10;
        t/=10;
    }
    reverse(r);
}
int main()
{
    int a,b;
    int r[N]={0};
    r[0]=1;
    len2++;
    scanf("%d %d",&a,&b);
    get_prim(a);
    int i=0;
    int j=0;
    for(i=0;i<len1;i++)
    {
        int p=prim[i];
        count[p]=get(a,p)-get(b,p)-get(a-b,p);
    }
    for(i=0;i<len1;i++)
    {
        int p=prim[i];
        for(j=0;j<count[p];j++)
        {
            mul(r,p);
        }
    }
    for(i=0;i<len2;i++)
    {
        printf("%d",r[i]);
    }
    return 0;
}
```
