# AcWing 789. 数的范围

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#define N 100010
int arr[N]={0};
int find_left(int*arr,int left,int right,int x)
{
    while(left<=right)
    {
        int mid=(right-left)/2+left;
        if(arr[mid]<x)
        {
            left=mid+1;
        }
        else if(arr[mid]>=x)
        {
            right=mid-1;
        }
    }
    return left;
}
int find_right(int*arr,int left,int right,int x)
{
    while(left<=right)
    {
        int mid=(right-left)/2+left;
        if(arr[mid]<=x)
        {
            left=mid+1;
        }
        else if(arr[mid]>x)
        {
            right=mid-1;
        }
    }
    return right;
}
int main()
{
    int n=0;
    int q=0;
    scanf("%d %d",&n,&q);
    int i=0;
    for(i=0;i<n;i++)
    {
        scanf("%d",&arr[i]);
    }
    while(q--)
    {
        int x=0;
        scanf("%d",&x);
        int l=find_left(arr,0,n-1,x);
        int r=find_right(arr,0,n-1,x);
        if(l<=r)
        {
            printf("%d %d\n",l,r);
        }
        else
        {
            printf("-1 -1\n");
        }
    }
    return 0;
}
```
