# AcWing 102. 最佳牛围栏

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
double cows[100010];
int n,m;
int check(double mid)
{
    double sum[100010]={0};
    int i,j;
    for(i=1;i<=n;i++)
    {
        sum[i]=sum[i-1]+cows[i]-mid;
    }
    double minv=0;
    for(i=0,j=m;j<=n;i++,j++)
    {
        minv=min(minv,sum[i]);
        if(sum[j]>=minv)
        {
            return 1;
        }
    }
    return 0;
}
int main()
{
    cin>>n>>m;
    int i;
    for(i=1;i<=n;i++)
    {
        cin>>cows[i];
    }
    double l=0;
    double r=2000;
    double mid=0;
    while(r-l>1e-5)
    {
        mid=(l+r)/2;
        if(check(mid))
        {
            l=mid;
        }
        else
        {
            r=mid;
        }
    }
    cout<<(int)(r*1000)<<endl;
    return 0;
}
```
