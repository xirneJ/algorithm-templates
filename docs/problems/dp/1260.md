# AcWing 1014. 登山

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
int main()
{
    int n;
    int i,j;
    cin>>n;
    int arr[1010]={0};
    int dp1[1010]={0};
    int dp2[1010]={0};
    for(i=0;i<n;i++)
    {
        cin>>arr[i];
        dp1[i]=1;
        dp2[i]=1;
    }
    int res=1;
    for(i=0;i<n;i++)
    {
        for(j=0;j<i;j++)
        {
            if(arr[i]>arr[j])
            {
                dp1[i]=max(dp1[i],dp1[j]+1);
            }
        }
    }
    for(i=n-1;i>=0;i--)
    {
        for(j=n-1;j>i;j--)
        {
            if(arr[i]>arr[j])
            {
                dp2[i]=max(dp2[i],dp2[j]+1);
            }
        }
    }
    for(i=0;i<n;i++)
    {
        if(arr[i]!=arr[i+1])
        {
            res=max(res,dp1[i]+dp2[i]-1);
        }
        if(i==(n-1))
        {
            res=max(res,dp1[i]);
        }
    }
    cout<<res<<endl;
    return 0;
}
```
