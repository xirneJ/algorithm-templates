# AcWing 187. 导弹防御系统

## 模板

```cpp
#include <iostream>
#include <algorithm>
#include <cstdlib>
using namespace std;
int arr[60];
int n;
int ups[60];
int downs[60];
int res;
void dps(int u,int ud,int dd)
{
    if(ud+dd>=res)
    {
        return;
    }
    if(u==n)
    {
        res=ud+dd;
        return;
    }
    int k=0;
    while(k<ud&&arr[u]<=ups[k])
    {
        k++;
    }
    int t=ups[k];
    ups[k]=arr[u];
    if(k==ud)
    {
        dps(u+1,ud+1,dd);
        ups[k]=t;
    }
    else
    {
        dps(u+1,ud,dd);
        ups[k]=t;
    }
    k=0;
    while(k<dd&&arr[u]>=downs[k])
    {
        k++;
    }
    t=downs[k];
    downs[k]=arr[u];
    if(k==dd)
    {
        dps(u+1,ud,dd+1);
        downs[k]=t;
    }
    else
    {
        dps(u+1,ud,dd);
        downs[k]=t;
    }
}
int main()
{
    while(scanf("%d",&n)&&n)
    {
        int i;
        for(i=0;i<n;i++)
        {
            cin>>arr[i];
        }
        res=n;
        dps(0,0,0);
        cout<<res<<endl;
    }
    return 0;
}
```
