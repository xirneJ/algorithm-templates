# AcWing 320. 能量项链

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
typedef long long ll;
int arr[210];
int sum[210];
int dp[210][210];
int main()
{
    int n;
    cin>>n;
    int i,j,k;
    arr[0]=1;
    for(i=1;i<=n;i++)
    {
        int w;
        cin>>w;
        arr[i]=w;
        arr[i+n]=w;
    }
    arr[2*n+1]=arr[1];
    for(i=1;i<=n;i++)
    {
        for(j=1;(j+i-1)<=2*n;j++)
        {
            int l=j;
            int r=j+i-1;
            for(k=l;k<r;k++)
            {
                dp[l][r]=max(dp[l][r],dp[l][k]+dp[k+1][r]+arr[l]*arr[k+1]*arr[r+1]);
            }
        }
    }
    int res=0;
    for(i=1;i<=n;i++)//做完记得取所有合法长度区间的最大值
    {
        res=max(res,dp[i][i+n-1]);
    }
    cout<<res<<endl;
    return 0;
}
```
