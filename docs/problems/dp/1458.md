# AcWing 135. 最大子序和

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
#define N 300010
int sum[N];
int team[N];
int arr[N];
int main()
{
    int n,m;
    cin>>n>>m;
    int i;
    for(i=1;i<=n;i++)
    {
        cin>>arr[i];
        sum[i]=sum[i-1]+arr[i];
    }
    int res=-2147483648;
    int left=0;
    int right=0;
    for(i=1;i<=n;i++)
    {
        while(team[left]<i-m)
        {
            left++;
        }
        res=max(res,sum[i]-sum[team[left]]);
        while(left<=right&&sum[team[right]]>sum[i])
        {
            right--;
        }
        team[++right]=i;
        // cout<<res<<' '<<sum[i]<<' '<<sum[team[left]]<<endl;
    }
    cout<<res<<endl;
    return 0;
}
```
