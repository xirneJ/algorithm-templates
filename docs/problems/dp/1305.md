# AcWing 1074. 二叉苹果树

## 模板

```cpp
#include <iostream>
#include <cstring>
#include <algorithm>
using namespace std;
#define N 110
int ne[N*2],np[N*2],h[N],idx,valus[N*2];
int dp[N][N];
void add(int a,int b,int c)
{
    np[idx]=b;
    ne[idx]=h[a];
    valus[idx]=c;
    h[a]=idx++;
}
int n,m;
void dps(int u,int fa)
{
    int i,j,k;
    for(i=h[u];i!=-1;i=ne[i])
    {
        int r=np[i];
        if(r==fa)continue;
        dps(r,u);
        for(j=m;j;j--)//不能从小到大，防止左脚猜右脚升天
        {
            for(k=0;k+1<=j;k++)
            {
                dp[u][j]=max(dp[u][j],dp[u][j-k-1]+valus[i]+dp[r][k]);
            }
        }
    }
}
int main()
{
    memset(h,-1,sizeof(h));
    cin>>n>>m;
    int i;
    for(i=1;i<n;i++)
    {
        int a,b,c;
        cin>>a>>b>>c;
        add(a,b,c);
        add(b,a,c);
    }
    dps(1,0);
    cout<<dp[1][m]<<endl;
    return 0;
}
```
