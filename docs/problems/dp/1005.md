# AcWing 897. 最长公共子序列

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#define N 1010
int dp[N][N]={0};
char arr1[N]={0};
char arr2[N]={0};
int max(int a,int b)
{
    return a>b?a:b;
}
int main()
{
    int len1=0;
    int len2=0;
    scanf("%d %d",&len1,&len2);
    int i=0;
    int j=0;
    scanf("%s %s",arr1+1,arr2+1);
    for(i=1;i<=len1;i++)
    {
        for(j=1;j<=len2;j++)
        {
            if(arr1[i]==arr2[j])
            {
                dp[i][j]=dp[i-1][j-1]+1;
            }
            else
            {
                dp[i][j]=max(dp[i-1][j],dp[i][j-1]);
            }
        }
    }
    printf("%d",dp[len1][len2]);
    return 0;
}
```
