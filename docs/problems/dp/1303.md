# AcWing 1073. 树的中心

## 模板

```cpp
#include <iostream>
#include <cstring>
#include <algorithm>
using namespace std;
#define N 10010
int d1[N],d2[N],up[N],p1[N],ne[2*N],np[2*N],h[N],idx,valus[N*2];
void add(int a,int b,int c)
{
    np[idx]=b;
    valus[idx]=c;
    ne[idx]=h[a];
    h[a]=idx++;
}
int n;
int d_dps(int u,int fa)
{
    int i;
    for(i=h[u];i!=-1;i=ne[i])
    {
        int r=np[i];
        if(r==fa)continue;
        int x=d_dps(r,u)+valus[i];
        if(x>=d1[u])//只能走一端进，防止一条路径同时更新d1和d2
        {
            d2[u]=d1[u];
            d1[u]=x;
            p1[u]=r;
        }
        else if(x>d2[u])
        {
            d2[u]=x;
        }
    }
    return d1[u];
}

void u_dps(int u,int fa)
{
    int i;
    for(i=h[u];i!=-1;i=ne[i])
    {
        int r=np[i];
        if(r==fa)continue;
        if(p1[u]==r)
        {
            up[r]=max(up[u],d2[u])+valus[i];
        }
        else
        {
            up[r]=max(up[u],d1[u])+valus[i];
        }
        u_dps(r,u);
    }
}
int main()
{
    memset(h,-1,sizeof(h));
    cin>>n;
    int i;
    for(i=1;i<n;i++)
    {
        int a,b,c;
        cin>>a>>b>>c;
        add(a,b,c);
        add(b,a,c);
    }
    int res=d_dps(1,0);
    u_dps(1,0);
    int ans=0x3f3f3f3f;
    for(i=1;i<=n;i++)
    {
        ans=min(ans,max(up[i],d1[i]));
    }
    cout<<ans<<endl;
    return 0;
}
```
