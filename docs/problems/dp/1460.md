# AcWing 1088. 旅行问题

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
#define N 1000010
typedef long long ll;
int oil[N*2];
int dis[N*2];
ll s[N*2];
int q[N*2];
bool ans[N];
int main()
{
    int n;
    cin>>n;
    int i;
    for(i=1;i<=n;i++)
    {
        scanf("%d %d",&oil[i],&dis[i]);
        s[i]=oil[i]-dis[i];
        s[n+i]=s[i];
    }
    for(i=1;i<=2*n;i++)
    {
        s[i]+=s[i-1];
    }
    int left=0;
    int right=0;
    q[0]=n+1;
    for(i=2*n;i>=0;i--)
    {
        while(q[left]>i+n)left++;
        if(i<n)
        {
            if(s[i]<=s[q[left]])ans[i+1]=true;
        }
        while(left<=right&&s[i]<=s[q[right]])right--;
        q[++right]=i;
    }
    left=0;
    right=0;
    q[0]=0;
    dis[0]=dis[n];
    for(i=1;i<=n;i++)
    {
        s[i]=oil[i]-dis[i-1];
        s[n+i]=s[i];
    }
    for(i=1;i<=2*n;i++)
    {
        s[i]+=s[i-1];
    }
    for(i=1;i<=n*2;i++)
    {
        while(q[left]<i-n)left++;
        if(i>n)
        {
            if(s[i]>=s[q[left]])ans[i-n]=true;
        }
        while(left<=right&&s[i]>=s[q[right]])right--;
        q[++right]=i;
    }
    for(i=1;i<=n;i++)
    {
        if(ans[i])
        {
            printf("TAK\n");
        }
        else
        {
            printf("NIE\n");
        }
    }
    return 0;
}
```
