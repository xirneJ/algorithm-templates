# AcWing 1082. 数字游戏

## 模板

```cpp
#include <iostream>
#include <vector>
#include <cstdio>
#include <algorithm>
using namespace std;
#define N 35
int f[N][N];
//有i位，最高位是j
int dp(int n)
{
    if(!n)return 1;
    vector<int>s;
    while(n)
    {
        s.push_back(n%10);
        n/=10;
    }
    n=s.size();
    int i,j;
    int res=0;
    int last=0;
    for(i=n-1;i>=0;i--)
    {
        int x=s[i];

        for(j=last;j<x;j++)
        {
            res+=f[i+1][j];
        }
        if(x<last)break;
        last=x;
        if(!i)res++;
    }
    return res;
}
int main()
{
    int i,j,k;
    for(i=0;i<10;i++)
    {
        f[1][i]=1;
    }
    for(k=2;k<=N;k++)
    {
        for(i=0;i<10;i++)
        {
            for(j=i;j<10;j++)
            {
                f[k][i]+=f[k-1][j];
            }
        }
    }
    int a,b;
    while(scanf("%d %d",&a,&b)==2)
    {
        if(a>b)swap(a,b);
        cout<<dp(b)-dp(a-1)<<endl;
    }
    return 0;
}
```
