# AcWing 1083. Windy数

## 模板

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
typedef long long ll;
#define N 15
int f[N][N];
int dp(int n)
{
    vector<int>s;
    while(n)
    {
        s.push_back(n%10);
        n/=10;
    }
    n=s.size();
    int i,j;
    int res=0;
    int last=-10;
    for(i=1;i<n;i++)
    {
        for(j=1;j<10;j++)
        {
            res+=f[i][j];
        }
    }
    for(i=n-1;i>=0;i--)
    {
        int x=s[i];
        if(i==n-1)
        {
            for(j=1;j<x;j++)
            {
                if(abs(last-j)>=2)res+=f[i+1][j];
            }
        }
        else
        {
            for(j=0;j<x;j++)
            {
                if(abs(last-j)>=2)res+=f[i+1][j];
            }
        }
        if(abs(x-last)<2)break;
        last=x;
        if(!i)res++;
    }
    return res;
}
int main()
{
    ll a,b;
    cin>>a>>b;
    if(a>b)swap(a,b);
    int i,j,k;
    for(i=0;i<10;i++)f[1][i]=1;
    for(k=2;k<=N;k++)
    {
        for(i=0;i<10;i++)
        {
            for(j=i-2;j>=0;j--)
            {
                f[k][i]+=f[k-1][j];
            }
            for(j=i+2;j<=9;j++)
            {
                f[k][i]+=f[k-1][j];
            }
        }
    }
    cout<<dp(b)-dp(a-1)<<endl;
    return 0;
}
```
