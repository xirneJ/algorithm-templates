# AcWing 1027. 方格取数

## 模板

```cpp
#include <iostream>
#include <algorithm>
#include <cstdio>
using namespace std;
int g[15][15];
int f[30][15][15];
int main()
{
    int i1,j1,i2,j2,n,k;
    cin>>n;
    int l,r,v;
    while(scanf("%d %d %d",&l,&r,&v)==3)
    {
        g[l][r]=v;
    }
    for(k=2;k<=n+n;k++)
    {
        for(i1=1;i1<=n;i1++)
        {
            for(i2=1;i2<=n;i2++)
            {
                j1=k-i1;
                j2=k-i2;
                if(j1>=1&&j1<=n&&j2>=1&&j2<=n)
                {
                    int t=g[i1][j1];
                    if(i1!=i2)
                    {
                        t+=g[i2][j2];
                    }
                    f[k][i1][i2]=max(f[k][i1][i2],f[k-1][i1-1][i2]+t);
                    f[k][i1][i2]=max(f[k][i1][i2],f[k-1][i1][i2-1]+t);
                    f[k][i1][i2]=max(f[k][i1][i2],f[k-1][i1-1][i2-1]+t);
                    f[k][i1][i2]=max(f[k][i1][i2],f[k-1][i1][i2]+t);
                }
            }
        }
    }
    cout<<f[n+n][n][n]<<endl;
    return 0;
}
```
