# AcWing 275. 传纸条

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
int g[60][60];
int dp[120][60][60];
int main()
{
    int n,m,i,j,k;
    cin>>n>>m;
    for(i=1;i<=n;i++)
    {
        for(j=1;j<=m;j++)
        {
            cin>>g[i][j];
        }
    }
    int i1,i2,j1,j2;
    for(k=2;k<=n+m;k++)
    {
        for(i1=1;i1<=n;i1++)
        {
            for(i2=1;i2<=n;i2++)
            {
                if(i1!=i2)
                {
                    int j1=k-i1;
                    int j2=k-i2;
                    if(j1>=1&&j1<=m&&j2>=1&&j2<=m)
                    {
                        int t=g[i1][j1];
                        t+=g[i2][j2];
                        int&x=dp[k][i1][i2];
                        x=max(x,dp[k-1][i1][i2]+t);
                        x=max(x,dp[k-1][i1-1][i2]+t);
                        x=max(x,dp[k-1][i1][i2-1]+t);
                        x=max(x,dp[k-1][i1-1][i2-1]+t);
                    }
                }
            }
        }
    }
    cout<<max(dp[n+m-1][n-1][n],dp[n+m-1][n][n-1])<<endl;
    return 0;
}
```
