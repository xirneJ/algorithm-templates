# AcWing 1022. 宠物小精灵之收服

## 模板

```cpp
#include <iostream>
#include <algorithm>
#include <cstring>
using namespace std;
struct node
{
    int s,c;
}arr[110];
int dp[1010][510];
int main()
{
    int NUM,T,n;
    cin>>NUM>>T>>n;
    int i,j,k;
    for(i=0;i<n;i++)
    {
        cin>>arr[i].c>>arr[i].s;
    }
    for(i=0;i<n;i++)
    {
        for(j=NUM;j>=arr[i].c;j--)
        {
            for(k=T-1;k>=arr[i].s;k--)
            {
                dp[j][k]=max(dp[j][k],dp[j-arr[i].c][k-arr[i].s]+1);
            }
        }
    }
    int res=T-1;
    while(res>0&&dp[NUM][res-1]==dp[NUM][T-1])
    {
        res--;
    }
    cout<<dp[NUM][T-1]<<' '<<T-res;
    return 0;
}
```
