# AcWing 1087. 修剪草坪

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
typedef long long ll;
#define N 100010
ll s[N];
ll f[N];
int q[N];
ll g(int i)
{
    if(!i)return 0;
    return f[i-1]-s[i];
}
ll max1(ll a,ll b)
{
    return a>b?a:b;
}
int main()
{
    int n,m;
    cin>>n>>m;
    int i;
    for(i=1;i<=n;i++)
    {
        cin>>s[i];
        s[i]+=s[i-1];
    }
    int left=0;
    int right=0;
    for(i=1;i<=n;i++)
    {
        while(q[left]<i-m)left++;
        f[i]=max1(f[i-1],g(q[left])+s[i]);
        while(left<=right&&g(i)>=g(q[right]))right--;//和队尾元素比
        q[++right]=i;
    }
    cout<<f[n]<<endl;
    return 0;
}
```
