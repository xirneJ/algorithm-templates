# AcWing 327. 玉米田

## 模板

```cpp
#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;
vector<int> st1;
int w[15];
vector<int> st2[1<<15];
int dp[15][1<<15];
int n,m;
bool check(int x)
{
    int i;
    for(i=0;i<m;i++)
    {
        if((x>>i&1)&&(x>>i+1&1))
        {
            return false;
        }
    }
    return true;
}
int main()
{
    cin>>n>>m;
    int i,j;
    for(i=0;i<n;i++)
    {
        int t=0;
        for(j=0;j<m;j++)
        {
            int s;
            cin>>s;
            t+=!s*(1<<j);
        }
        w[i]=t;
    }

    for(i=0;i<(1<<m);i++)
    {
        if(check(i))
        {
            st1.push_back(i);
        }
    }

    for(i=0;i<st1.size();i++)
    {
        for(j=0;j<st1.size();j++)
        {
            int a=st1[i];
            int b=st1[j];
            if((a&b)==0)
            {
                st2[i].push_back(b);
            }
        }
    }

    for(i=0;i<st1.size();i++)
    {
        if(!(w[0]&st1[i]))
        {
            dp[0][st1[i]]=1;
        }
    }
    for(i=1;i<n;i++)
    {
        for(j=0;j<st1.size();j++)
        {
            for(int k:st2[j])
            {
                if((w[i]&st1[j])==0)
                {
                    dp[i][st1[j]]=(dp[i][st1[j]]+dp[i-1][k])%100000000;
                }
            }
        }
    }
    int res=0;
    for(i=0;i<st1.size();i++)
    {
        if(!(w[n-1]&st1[i]))
        {
            res=(res+dp[n-1][st1[i]])%100000000;
        }
    }
    cout<<res<<endl;
    return 0;
}
```
