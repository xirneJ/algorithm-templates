# AcWing 1013. 机器分配

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
int g[20][20];
int dp[20][20];
int main()
{
    int n,m;
    int i,j,k;
    cin>>n>>m;
    for(i=1;i<=n;i++)
    {
        for(j=1;j<=m;j++)
        {
            cin>>g[i][j];
        }
    }
    for(i=n;i>=1;i--)
    {
        for(j=0;j<=m;j++)
        {
            for(k=0;k<=j;k++)
            {
                dp[i][j]=max(dp[i][j],dp[i+1][j-k]+g[i][k]);
            }
        }
    }
    cout<<dp[1][m]<<endl;
    for(i=1;i<=n;i++)
    {
        for(j=0;j<=m;j++)
        {
            if(dp[i][m]==(dp[i+1][m-j]+g[i][j]))//是用剩下的比较，我这里犯了个错误，原来是dp[i][j],实际上是dp[i][m];
            {
                cout<<i<<' '<<j<<endl;
                m-=j;
                break;
            }
        }
    }

    return 0;
}
```
