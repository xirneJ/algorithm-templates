# AcWing 285. 没有上司的舞会

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#define N 6010
int h[N]={0};
int np[N]={0};
int ne[N]={0};
int f[N]={0};
int idx=0;
int happy[N]={0};
int dp[N][2]={0};
void add(int a,int b)
{
    np[idx]=b;
    ne[idx]=h[a];
    h[a]=idx++;
}
int max(int a,int b)
{
    return a>b?a:b;
}
void dps(int u)
{
    dp[u][1]=happy[u];
    int i=0;
    int j=0;
    for(i=h[u];i!=-1;i=ne[i])
    {
        int son=np[i];
        dps(son);
        dp[u][1]+=dp[son][0];
        dp[u][0]+=max(dp[son][0],dp[son][1]);
    }
}
int main()
{
    int n=0;
    scanf("%d",&n);
    memset(h,-1,sizeof(h));
    int i=0;
    for(i=1;i<=n;i++)
    {
        scanf("%d",&happy[i]);
    }
    for(i=0;i<n-1;i++)
    {
        int s=0;
        int p=0;
        scanf("%d %d",&s,&p);
        add(p,s);
        f[s]=1;
    }
    int root=0;
    for(i=1;i<=n;i++)
    {
        if(f[i]==0)
        {
            root=i;
            break;
        }
    }
    dps(root);
    printf("%d",max(dp[root][1],dp[root][0]));
    return 0;
}
```
