# AcWing 1068. 环形石子合并

## 模板

```cpp
#include <iostream>
#include <algorithm>
#include <cstring>
using namespace std;
int arr[410];
int sum[410];
int min_dp[410][410];
int max_dp[410][410];
int main()
{
    memset(min_dp,0x3f3f,sizeof(min_dp));
    int n;
    cin>>n;
    int i,j,k;
    for(i=1;i<=n;i++)
    {
        int w;
        cin>>w;
        arr[i]=w;
        arr[i+n]=w;
    }
    for(i=1;i<=2*n;i++)
    {
        sum[i]=sum[i-1]+arr[i];
    }
    for(i=1;i<=n*2;i++)
    {
        min_dp[i][i]=0;
    }

    for(i=1;i<=n;i++)
    {
        for(j=1;(j+i-1)<=2*n;j++)
        {
            int l=j;
            int r=j+i-1;
            for(k=l;k<r;k++)
            {
                min_dp[l][r]=min(min_dp[l][r],min_dp[l][k]+min_dp[k+1][r]+sum[r]-sum[l-1]);
                max_dp[l][r]=max(max_dp[l][r],max_dp[l][k]+max_dp[k+1][r]+sum[r]-sum[l-1]);
            }
        }
    }
    int min_res=0x3f3f3f;
    int max_res=0;
    for(i=1;(i+n-1)<=2*n;i++)
    {
        min_res=min(min_res,min_dp[i][i+n-1]);
        max_res=max(max_res,max_dp[i][i+n-1]);
    }
    cout<<min_res<<endl;
    cout<<max_res<<endl;
    return 0;
}
```
