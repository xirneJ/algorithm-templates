# AcWing 1015. 摘花生

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--)
    {
        int g[110][110];
        int dp[110][110];
        int i,j;
        int n,m;
        cin>>n>>m;
        for(i=0;i<n;i++)
        {
            for(j=0;j<m;j++)
            {
                cin>>g[i][j];
            }
        }
        dp[0][0]=g[0][0];
        for(i=1;i<n;i++)
        {
            dp[i][0]=dp[i-1][0]+g[i][0];
        }
        for(i=1;i<m;i++)
        {
            dp[0][i]=dp[0][i-1]+g[0][i];
        }
        for(i=1;i<n;i++)
        {
            for(j=1;j<m;j++)
            {
                dp[i][j]=max(dp[i-1][j],dp[i][j-1])+g[i][j];
            }
        }
        cout<<dp[n-1][m-1]<<endl;
    }
    return 0;
}
```
