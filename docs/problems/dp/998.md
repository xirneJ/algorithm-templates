# AcWing 3. 完全背包问题

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
typedef long long ll;
#define N 1010
struct arry
{
    int v;
    int valu;
}arr[N];
int dp[N];
int main()
{
    int n,V;
    cin>>n>>V;
    int i,j;
    for(i=0;i<n;i++)
    {
        cin>>arr[i].v;
        cin>>arr[i].valu;
    }
    for(i=0;i<n;i++)
    {
        for(j=arr[i].v;j<=V;j++)
        {
            dp[j]=max(dp[j],dp[j-arr[i].v]+arr[i].valu);
        }
    }
    cout<<dp[V]<<endl;
    return 0;
}
```
