# AcWing 902. 最短编辑距离

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#define N 1010
char arr1[N]={0};
char arr2[N]={0};
int dp[N][N]={0};
int max(int a,int b)
{
    return a>b?a:b;
}
int min(int a,int b)
{
    return a<b?a:b;
}
int main()
{
    int len1=0;
    scanf("%d %s",&len1,arr1+1);
    int len2=0;
    scanf("%d %s",&len2,arr2+1);
    int i=0;
    int j=0;
    for(i=1;i<=len1;i++)
    {
        dp[i][0]=i;
    }
    for(i=1;i<=len2;i++)
    {
        dp[0][i]=i;
    }
    for(i=1;i<=len1;i++)
    {
        for(j=1;j<=len2;j++)
        {
            if(arr1[i]==arr2[j])
            {
                dp[i][j]=dp[i-1][j-1];
            }
            else
            {
                dp[i][j]=min(min(dp[i-1][j],dp[i][j-1]),dp[i-1][j-1])+1;
            }
        }
    }
    printf("%d",dp[len1][len2]);
    return 0;
}
```
