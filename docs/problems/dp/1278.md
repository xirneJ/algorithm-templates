# AcWing 1020. 潜水员

## 模板

```cpp
#include <iostream>
#include <cstring>
#include <algorithm>
using namespace std;
struct node
{
    int o2,n2,w;
}arr[1010];
int dp[30][80];
int main()
{
    int O2,N2;
    cin>>O2>>N2;
    int i,j,k;
    int n;
    cin>>n;
    for(i=0;i<n;i++)
    {
        cin>>arr[i].o2>>arr[i].n2>>arr[i].w;
    }
    memset(dp,0x3f3f3f3f,sizeof(dp));
    dp[0][0]=0;
    for(i=0;i<n;i++)
    {
        for(j=O2;j>=0;j--)
        {
            for(k=N2;k>=0;k--)
            {
               dp[j][k]=min(dp[j][k],dp[max(0,j-arr[i].o2)][max(0,k-arr[i].n2)]+arr[i].w);
            }
        }
    }
    cout<<dp[O2][N2]<<endl;
    return 0;
}
```
