# AcWing 487. 金明的预算方案

## 模板

```cpp
#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;
struct node
{
    int p,imp,valu;
};
struct ll
{
    int p,imp,fa;
}arr[100];
int dp[40000];
vector<node> s[100];
int pa[100];
int len=0;
int main()
{
    int n,m;
    cin>>m>>n;
    int i,j;
    for(i=1;i<=n;i++)
    {
        cin>>arr[i].p>>arr[i].imp>>arr[i].fa;
    }
    for(i=1;i<=n;i++)
    {
        if(arr[i].fa==0)
        {
            s[i].push_back({arr[i].p,arr[i].imp,arr[i].p*arr[i].imp});
            pa[len]=i;
            len++;
        }
    }
    for(i=1;i<=n;i++)
    {
        if(arr[i].fa!=0)
        {
            vector<node>s1(s[arr[i].fa]);
            for(auto it=s1.begin();it!=s1.end();it++)
            {
                s[arr[i].fa].push_back({arr[i].p+it->p,arr[i].imp+it->imp,it->valu+arr[i].p*arr[i].imp});
            }
        }
    }
    for(i=0;i<len;i++)
    {
        for(j=m;j>=0;j--)
        {
            for(auto it=s[pa[i]].begin();it!=s[pa[i]].end();it++)
            {
                if(j>=it->p)
                {
                    dp[j]=max(dp[j],dp[j-it->p]+it->valu);
                }
            }
        }
    }
    cout<<dp[m]<<endl;
    return 0;
}
```
