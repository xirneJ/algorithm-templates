# AcWing 532. 货币系统

## 模板

```cpp
#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--)
    {
        int n;
        int arr[110];
        int dp[25010]={0};
        cin>>n;
        int i,j;
        dp[0]=1;
        int res=0;
        for(i=0;i<n;i++)
        {
            cin>>arr[i];
        }
        sort(arr,arr+n);
        int m=arr[n-1];
        for(i=0;i<n;i++)//类似于筛质数，用数组里面的数把后面能表示出来的数筛掉
        {
            if(!dp[arr[i]])
            {
                res++;
            }
            for(j=arr[i];j<=m;j++)
            {
                dp[j]|=dp[j-arr[i]];
            }
        }
        cout<<res<<endl;
    }
    return 0;
}
```
