# AcWing 479. 加分二叉树

## 模板

```cpp
#include <iostream>
#include <algorithm>
#include <cstdlib>
using namespace std;
int tree[35];
int dp[35][35];
int g[35][35];
int n;
void print(int l,int r)
{
    if(l>r)
    {
        return ;
    }
    else
    {
        int root=g[l][r];
        cout<<root<<' ';
        print(l,root-1);
        print(root+1,r);
    }
}
int main()
{
    cin>>n;
    int i,j,k;
    for(i=1;i<=n;i++)
    {
        cin>>tree[i];
        dp[i][i]=tree[i];
        g[i][i]=i;
    }
    for(i=2;i<=n;i++)
    {
        for(j=1;j+i-1<=n;j++)
        {
            int l=j;
            int r=i+j-1;
            for(k=l;k<=r;k++)
            {
                if(k==l)
                {
                    if(dp[l][r]<dp[k+1][r]+tree[k])
                    {
                        dp[l][r]=dp[k+1][r]+tree[k];
                        g[l][r]=k;
                    }
                }
                else if(k==r)
                {
                    if(dp[l][r]<dp[l][k-1]+tree[k])
                    {
                        dp[l][r]=dp[l][k-1]+tree[k];
                        g[l][r]=k;
                    }
                }
                else
                {
                    if(dp[l][r]<(dp[l][k-1]*dp[k+1][r]+tree[k]))
                    {
                        dp[l][r]=max(dp[l][r],dp[l][k-1]*dp[k+1][r]+tree[k]);
                        g[l][r]=k;
                    }
                }
            }
        }
    }
    cout<<dp[1][n]<<endl;
    print(1,n);
    return 0;
}
```
