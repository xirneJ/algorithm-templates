# AcWing 1064. 小国王

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
typedef long long ll;
int arr[11][11];
int st[1<<10];
ll dp[11][1<<10][110];
typedef pair<int,int>pii;
vector<pii>s;
int main()
{
    int n,m;
    cin>>n>>m;
    int i,j;
    for(i=0;i<(1<<n);i++)
    {
        st[i]=1;
        int c=0;
        for(j=0;j<n;j++)
        {
            if(i>>j&1)
            {
                c++;
            }
            else
            {
                if(c>1)
                {
                    st[i]=0;
                }
                c=0;
            }
        }
        if(c>1)
        {
            st[i]=0;
        }
    }

    for(i=0;i<(1<<n);i++)
    {
        if(st[i])
        {
            int c=0;
            for(j=0;j<n;j++)
            {
                if(i>>j&1)
                {
                    c++;
                }
            }
            s.push_back({i,c});
        }
    }

    for(auto it=s.begin();it!=s.end();it++)
    {
        dp[0][it->first][it->second]=1;
    }

    ll res=0;
    for(i=1;i<n;i++)
    {
        for(j=0;j<=m;j++)
        {
            for(auto a=s.begin();a!=s.end();a++)
            {
                for(auto b=s.begin();b!=s.end();b++)
                {
                    if(st[a->first|b->first]&&j>=a->second&&(a->first&b->first)==0)//如果没有a->first&b->first这个001和001的状态检查是对的，实际上并不对
                    {
                        dp[i][a->first][j]+=dp[i-1][b->first][j-a->second];
                    }
                }
            }
        }
    }
    for(auto a=s.begin();a!=s.end();a++)
    {
        res+=dp[n-1][a->first][m];
    }
    cout<<res<<endl;
    return 0;
}
```
