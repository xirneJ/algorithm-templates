# AcWing 1016. 最大上升子序列和

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
int arr[1010];
int dp[1010];
int main()
{
    int n,i,j;
    cin>>n;
    for(i=0;i<n;i++)
    {
        cin>>arr[i];
    }
    int res=0;
    for(i=0;i<n;i++)
    {
        dp[i]=arr[i];
        res=max(res,dp[i]);
    }
    for(i=0;i<n;i++)
    {
        for(j=0;j<i;j++)
        {
            if(arr[j]<arr[i])
            {
                dp[i]=max(dp[i],dp[j]+arr[i]);
                res=max(res,dp[i]);
            }
        }
    }
    cout<<res<<endl;
    return 0;
}
```
