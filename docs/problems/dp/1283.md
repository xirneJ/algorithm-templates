# AcWing 12. 背包问题求具体方案

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
struct node
{
    int v,valu;
}arr[1010];
int dp[1010][1010];
int main()
{
    int n,m;
    cin>>n>>m;
    int i,j;
    for(i=1;i<=n;i++)
    {
        cin>>arr[i].v>>arr[i].valu;
    }
    for(i=n;i>=1;i--)
    {
        for(j=m;j>=0;j--)
        {
            if(j>=arr[i].v)
            {
                dp[i][j]=max(dp[i+1][j],dp[i+1][j-arr[i].v]+arr[i].valu);
            }
            else
            {
                dp[i][j]=dp[i+1][j];
            }
        }
    }
    j=m;
    for(i=1;i<=n;i++)
    {
        if(j>=arr[i].v&&dp[i][j]==(dp[i+1][j-arr[i].v]+arr[i].valu))
        {
            cout<<i<<' ';
            j-=arr[i].v;
        }
    }
    return 0;
}
```
