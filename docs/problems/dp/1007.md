# AcWing 282. 石子合并

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#define N 301
#define MAX 0x3f3f3f3f
int dp[N][N]={0};
int arr[N]={0};
int sum[N]={0};
int min(int a,int b)
{
    return a<b?a:b;
}
int main()
{
    int n=0;
    int i=0;
    int j=0;
    int k=0;
    scanf("%d",&n);
    for(i=0;i<n;i++)
    {
        scanf("%d",&arr[i]);
    }
    sum[0]=arr[0];
    for(i=0;i<n;i++)
    {
        for(j=0;j<n;j++)
        {
            if(i==j)
            {
                dp[i][j]=0;
            }
            else
            {
                dp[i][j]=MAX;
            }
        }
    }
    for(i=1;i<n;i++)
    {
        sum[i]=sum[i-1]+arr[i];
    }
    for(k=2;k<=n;k++)
    {
        for(i=0;i<n&&i+k-1<n;i++)
        {
            for(j=i;j<i+k-1;j++)
            {
                if(i==0)
                {
                    dp[i][i+k-1]=min(dp[i][i+k-1],dp[i][j]+dp[j+1][i+k-1]+sum[i+k-1]);
                }
                else if(i>=1)
                {
                    dp[i][i+k-1]=min(dp[i][i+k-1],dp[i][j]+dp[j+1][i+k-1]+sum[i+k-1]-sum[i-1]);
                }
            }
        }
    }
    printf("%d",dp[0][n-1]);
    return 0;
}
```
