# AcWing 524. 愤怒的小鸟

## 模板

```cpp
#include<iostream>
#include<algorithm>
#include <cmath>
#include <cstring>
using namespace std;
#define x first
#define y second
typedef pair<double,double>pdd;
int cmp(double a,double b)
{
    if(fabs(a-b)<1e-8)
    {
        return 0;
    }
    if(a<b)
    {
        return -1;
    }
    else
    {
        return 1;
    }
}
int dp[1<<18];
int main()
{
    int t;
    cin>>t;
    while(t--)
    {
        memset(dp,0x3f3f,sizeof(dp));
        int n,m;
        cin>>n>>m;
        int path[20][20]={0};
        vector<pdd>s;
        int i,j,k;
        for(i=0;i<n;i++)
        {
            double x,y;
            cin>>x>>y;
            s.push_back({x,y});
        }

        for(i=0;i<n;i++)
        {
            path[i][i]=1<<i;
            for(j=0;j<n;j++)
            {
                double x1=s[i].x;
                double x2=s[j].x;
                double y1=s[i].y;
                double y2=s[j].y;
                double a =(y1/x1-y2/x2)/(x1-x2);
                double b =y1/x1-a*x1;
                if(cmp(a,0)>=0) continue;//a不能等于0而且a也不能大于0
                if(!cmp(x1,x2))continue;
                int t=0;
                for(k=0;k<n;k++)
                {
                    double x3=s[k].x;
                    double y3=s[k].y;
                    if(!cmp(a*x3*x3+b*x3,y3))
                    {
                        t+=1<<k;
                    }
                }
                path[i][j]=t;
            }
        }

        dp[0]=0;
        for(i=0;i<(1<<n);i++)
        {
            int t=0;
            for(j=0;j<n;j++)
            {
                if(!(i>>j&1))
                {
                    t=j;
                    break;
                }
            }
            for(k=0;k<n;k++)
            {
                dp[i|path[t][k]]=min(dp[i|path[t][k]],dp[i]+1);
            }
        }
        cout<<dp[(1<<n)-1]<<endl;
    }
    return 0;
}
```
