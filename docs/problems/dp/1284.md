# AcWing 734. 能量石

## 模板

```cpp
#include <iostream>
#include <algorithm>
#include <cstring>
#include <cstdio>
using namespace std;
struct node
{
    int t,e,l;
}arr[100];
bool cmp(node a,node b)
{
    return a.t*b.l<a.l*b.t;
}
int dp[10010];
int main()
{
    int T,t;
    cin>>T;
    for(t=1;t<=T;t++)
    {
        memset(dp,0,sizeof(dp));//记得重置dp数组
        int n,i,j;
        cin>>n;
        int m=0;
        for(i=0;i<n;i++)
        {
            cin>>arr[i].t>>arr[i].e>>arr[i].l;
            m+=arr[i].t;
        }
        sort(arr,arr+n,cmp);
        for(i=0;i<n;i++)
        {
            for(j=m;j>=arr[i].t;j--)
            {
                dp[j]=max(dp[j],dp[j-arr[i].t]+max(0,arr[i].e-arr[i].l*(j-arr[i].t)));
            }
        }
        int res=0;
        for(i=1;i<=m;i++)
        {
            res=max(res,dp[i]);
        }
        printf("Case #%d: %d\n",t,res);
    }
    return 0;
}
```
