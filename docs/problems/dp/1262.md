# AcWing 1012. 友好城市

## 模板

```cpp
#include <algorithm>
#include <iostream>
#include <cstdlib>
using namespace std;
int dp1[5010];
int dp2[5010];
struct node
{
    int x,y;
}arr[5010];
int cmp(const void*e1,const void*e2)
{
    node*a1=(node*)e1;
    node*a2=(node*)e2;
    return a1->x-a2->x;
}
int main()
{
    int n,i,j;
    cin>>n;
    for(i=0;i<n;i++)
    {
        cin>>arr[i].x>>arr[i].y;
    }
    qsort(arr,n,sizeof(node),cmp);//本体要记得排序
    for(i=0;i<n;i++)
    {
        dp1[i]=1;
        dp2[i]=1;
    }
    int res=0;
    for(i=0;i<n;i++)
    {
        for(j=0;j<i;j++)
        {
            if(arr[j].x<=arr[i].x&&arr[j].y<=arr[i].y)
            {
                dp1[i]=max(dp1[i],dp1[j]+1);
                res=max(res,dp1[i]);
            }
            if(arr[j].x>=arr[i].x&&arr[j].y>=arr[i].y)
            {
                dp2[i]=max(dp2[i],dp2[j]+1);
                res=max(res,dp2[i]);
            }
        }
    }
    cout<<res<<endl;
    return 0;
}
```
