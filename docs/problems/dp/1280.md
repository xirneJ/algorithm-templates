# AcWing 426. 开心的金明

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
struct node
{
    int cost,valu;
}arr[30];
int dp[30010];
int main()
{
    int n,m;
    cin>>m>>n;
    int i,j;
    for(i=0;i<n;i++)
    {
        cin>>arr[i].cost>>arr[i].valu;
    }
    for(i=0;i<n;i++)
    {
        for(j=m;j>=arr[i].cost;j--)
        {
            dp[j]=max(dp[j],dp[j-arr[i].cost]+arr[i].valu*arr[i].cost);
        }
    }
    cout<<dp[m]<<endl;
    return 0;
}
```
