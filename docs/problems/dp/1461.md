# AcWing 1089. 烽火传递

## 模板

```cpp
#include <iostream>
#include <cstring>
#include <algorithm>
using namespace std;
#define N 200010
int f[N];
int arr[N];
int q[N];
int main()
{
    int n,m;
    cin>>n>>m;
    int i;
    for(i=1;i<=n;i++)
    {
        scanf("%d",&arr[i]);
    }
    int left=0;
    int right=0;
    for(i=1;i<=n;i++)
    {
        while(q[left]<i-m)left++;
        f[i]=f[q[left]]+arr[i];//从前面最小的继承过来
        while(left<=right&&f[q[right]]>=f[i])right--;
        q[++right]=i;
    }
    int ans=0x3f3f3f3f;
    for(i=n-m+1;i<=n;i++)//不能从1开始，要从n-m+1开始
    {
        ans=min(ans,f[i]);
    }
    cout<<ans<<endl;
    return 0;
}
```
