# AcWing 11. 背包问题求方案数

## 模板

```cpp
#include <iostream>
#include <algorithm>
#include <cstring>
using namespace std;
struct node
{
    int v,valu;
}arr[1010];
int dp[1010];
int g[1010];
int main()
{
    int n,m;
    cin>>n>>m;
    int i,j;
    for(i=0;i<n;i++)
    {
        cin>>arr[i].v>>arr[i].valu;
    }
    g[0]=1;
    for(i=0;i<n;i++)
    {
        for(j=m;j>=arr[i].v;j--)
        {
            if(dp[j]<(dp[j-arr[i].v]+arr[i].valu))
            {
                dp[j]=dp[j-arr[i].v]+arr[i].valu;//如果dp的值能更新，代表第一次到这里，g值直接继承前面的值
                g[j]=g[j-arr[i].v];
            }
            else if(dp[j]==(dp[j-arr[i].v]+arr[i].valu))
            {
                g[j]=(g[j]+g[j-arr[i].v])%1000000007;//如果两者相等，g值直接加上前面的值
            }
        }
    }
    int res=0;
    for(i=0;i<=m;i++)
    {
        if(dp[i]>dp[res])
        {
            res=i;
        }
    }
    int ans=0;
    for(i=0;i<=m;i++)
    {
        if(dp[i]==dp[res])//题目说的是不超过背包体积，所有最大值可能不再dp[m]那里，这道题比较巧，最大值刚好在dp[m]那，所有这里直接等于dp[m]就可以了
        {
            ans+=g[i];
        }
    }
    cout<<ans<<endl;
    return 0;
}
```
