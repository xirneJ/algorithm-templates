# AcWing 291. 蒙德里安的梦想

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
typedef long long ll;
int main()
{
    int n=0;
    int m=0;
    int i=0;
    int j=0;
    int k=0;
    while(scanf("%d %d",&n,&m)&&(n||m))
    {
        ll dp[15][3000]={0};
        int sym[3000]={0};
        for(i=0;i<(1<<n);i++)
        {
            int count=0;
            sym[i]=1;
            for(j=0;j<n;j++)
            {
                if(i>>j&1)
                {
                    if(count&1)
                    {
                        sym[i]=0;
                    }
                    count=0;
                }
                else
                {
                    count++;
                }
            }
            if(count&1)
            {
                sym[i]=0;
            }
        }
        dp[0][0]=1;
        for(i=1;i<=m;i++)
        {
            for(j=0;j<(1<<n);j++)
            {
                for(k=0;k<(1<<n);k++)
                {
                    if((j&k)==0&&sym[j|k])
                    {
                        dp[i][j]+=dp[i-1][k];
                    }
                }
            }
        }
        printf("%lld\n",dp[m][0]);
    }
    return 0;
}
```
