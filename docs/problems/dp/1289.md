# AcWing 1058. 股票买卖 V

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
int arr[100010];
int dp[5][100010];
//0表示空状态
//1代表买入股票
//2代表持有股票
//3代表卖出股票
//4代表冷冻期
int main()
{
    int n;
    cin>>n;
    int i,j;
    for(i=0;i<n;i++)
    {
        cin>>arr[i];
    }
    dp[1][0]=-arr[0];
    dp[2][0]=-0x3f3f3f;
    dp[3][0]=-0x3f3f3f;
    dp[4][0]=-0x3f3f3f;
    for(i=1;i<n;i++)
    {
        dp[0][i]=max(dp[0][i-1],dp[4][i-1]);
        dp[1][i]=max(dp[0][i-1],dp[4][i-1])-arr[i];
        dp[2][i]=max(dp[1][i-1],dp[2][i-1]);
        dp[3][i]=max(dp[1][i-1],dp[2][i-1])+arr[i];
        dp[4][i]=dp[3][i-1];
    }
    ;

    dp[4][n-1];
    cout<<max(max(max(dp[0][n-1],dp[1][n-1]), max(dp[2][n-1],dp[3][n-1])),dp[4][n-1]);
    return 0;
}
```
