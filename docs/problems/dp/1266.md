# AcWing 272. 最长公共上升子序列

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
int a[3010];
int b[3010];
int n;
int dp[3010][3010];
int main()
{
    cin>>n;
    int i,j,k;
    for(i=1;i<=n;i++)
    {
        cin>>a[i];
    }
    for(i=1;i<=n;i++)
    {
        cin>>b[i];
    }
    for(i=1;i<=n;i++)
    {
        int maxv=1;
        for(j=1;j<=n;j++)
        {
            dp[i][j]=dp[i-1][j];
            if(a[i]==b[j])
            {
                dp[i][j]=max(dp[i][j],maxv);
            }
            if(b[j]<a[i])
            {
                maxv=max(maxv,dp[i][j]+1);
            }
        }
    }
    int res=0;
    for(i=1;i<=n;i++)
    {
        res=max(res,dp[n][i]);
    }
    cout<<res<<endl;
    return 0;
}
```
