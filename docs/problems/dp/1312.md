# AcWing 1085. 不要62

## 模板

```cpp
#include <iostream>
#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;
#define N 15
int f[N][N];
int dp(int n)
{
    if(!n)return 0;
    vector<int>s;
    while(n)
    {
        s.push_back(n%10);
        n/=10;
    }
    n=s.size();
    int i,j;
    int res=0;
    for(i=1;i<n;i++)
    {
        for(j=1;j<10;j++)
        {
            res+=f[i][j];
        }
    }
    int last=0;
    for(i=n-1;i>=0;i--)
    {
        int x=s[i];
        if(i==n-1)
        {
            for(j=1;j<x;j++)
            {
                res+=f[i+1][j];
            }
        }
        else
        {
            for(j=0;j<x;j++)
            {
                if(last==6&&j==2)continue;
                res+=f[i+1][j];
            }
        }
        if(x==4)break;
        if(last==6&&x==2)break;
        last=x;
        if(!i)res++;
    }
    return res;
}
int main()
{
    int a,b;
    int i,j,k;
    for(i=0;i<10;i++)
    {
        f[1][i]=1;
    }
    f[1][4]=0;
    for(k=2;k<N;k++)
    {
        for(i=0;i<10;i++)
        {
            if(i==4)continue;
            for(j=0;j<10;j++)
            {
                if(i==6&&j==2)continue;
                f[k][i]+=f[k-1][j];
            }
        }
    }
    while(scanf("%d %d",&a,&b)==2&&(a||b))
    {
        if(a>b)swap(a,b);
        cout<<dp(b)-dp(a-1)<<endl;
    }
    return 0;
}
```
