# AcWing 91. 最短Hamilton路径

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#define N 25
#define MAX 0x3f3f3f
int dis[N][N]={0};
int dp[N][1<<20]={0};
int min(int a,int b)
{
    return a<b?a:b;
}
int main()
{
    int n=0;
    scanf("%d",&n);
    int i=0;
    int j=0;
    int k=0;
    for(i=0;i<n;i++)
    {
        for(j=0;j<n;j++)
        {
            scanf("%d",&dis[i][j]);
        }
    }
    memset(dp,MAX,sizeof(dp));
    dp[0][1]=0;
    for(i=0;i<(1<<n);i++)
    {
        for(j=0;j<n;j++)
        {
            if(i>>j&1)
            {
                for(k=0;k<n;k++)
                {
                    if(i-(1<<j&1)>>k)
                    {
                        dp[j][i]=min(dp[j][i],dp[k][i-(1<<j)]+dis[k][j]);
                    }
                }
            }
        }
    }
    printf("%d",dp[n-1][(1<<n)-1]);
    return 0;
}
```
