# AcWing 9. 分组背包问题

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
#define N 110
int v[N][N];
int valu[N][N];
int cnt[N];
int dp[N];
int main()
{
    int n,V;
    cin>>n>>V;
    int i,j,k;
    for(i=0;i<n;i++)
    {
        cin>>cnt[i];
        for(j=0;j<cnt[i];j++)
        {
            cin>>v[i][j];
            cin>>valu[i][j];
        }
    }
    for(i=0;i<n;i++)
    {
        for(j=V;j>=0;j--)
        {
            for(k=0;k<cnt[i];k++)
            {
                if(j>=v[i][k])
                {
                    dp[j]=max(dp[j],dp[j-v[i][k]]+valu[i][k]);
                }
            }
        }
    }
    cout<<dp[V]<<endl;
    return 0;
}
```
