# AcWing 1077. 皇宫看守

## 模板

```cpp
#include <iostream>
#include <cstring>
#include <algorithm>
using namespace std;
#define N 1510
int dp[N][3];
//0表示被父节点看到
//1表示被子节点看到
//2表示有守卫
int ne[N*2];
int np[N*2];
int h[N];
int w[N];
int idx;
void add(int a,int b)
{
    np[idx]=b;
    ne[idx]=h[a];
    h[a]=idx++;
}
bool st[N];
int n;
int dps(int u)
{
    dp[u][0]=0;
    dp[u][1]=0;
    dp[u][2]=w[u];
    int i;
    int sum=0;
    for(i=h[u];i!=-1;i=ne[i])
    {
        int r=np[i];
        dps(r);
        dp[u][0]+=min(dp[r][1],dp[r][2]);
        dp[u][2]+=min(min(dp[r][0],dp[r][1]),dp[r][2]);
        sum+=min(dp[r][2],dp[r][1]);
    }
    dp[u][1]=0x3f3f3f3f;
    for(i=h[u];i!=-1;i=ne[i])
    {
        int r=np[i];
        dp[u][1]=min(dp[u][1],sum-min(dp[r][2],dp[r][1])+dp[r][2]);
    }
}
int main()
{
    memset(h,-1,sizeof(h));
    memset(dp,0x3f3f3f3f,sizeof(dp));
    cin>>n;
    int i,j;
    for(i=1;i<=n;i++)
    {
        int l,m;
        cin>>l>>w[l]>>m;
        for(j=0;j<m;j++)
        {
            int r;
            cin>>r;
            add(l,r);
            st[r]=true;
        }
    }
    int root=0;
    for(i=1;i<=n;i++)
    {
        if(!st[i])
        {
            root=i;
            break;
        }
    }
    dps(root);
    cout<<min(dp[root][1],dp[root][2])<<endl;
    return 0;
}
```
