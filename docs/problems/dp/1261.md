# AcWing 482. 合唱队形

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
int arr[110];
int main()
{
    int i,j;
    int n;
    cin>>n;
    for(i=0;i<n;i++)
    {
        cin>>arr[i];
    }
    int dp1[110]={0};
    int dp2[110]={0};
    for(i=0;i<n;i++)
    {
        dp1[i]=1;
        dp2[i]=1;
    }
    for(i=0;i<n;i++)
    {
        for(j=0;j<i;j++)
        {
            if(arr[i]>arr[j])
            {
                dp1[i]=max(dp1[i],dp1[j]+1);
            }
        }
    }
    for(i=n-1;i>=0;i--)
    {
        for(j=n-1;j>i;j--)
        {
            if(arr[i]>arr[j])
            {
                dp2[i]=max(dp2[i],dp2[j]+1);
            }
        }
    }
    int res=1;
    for(i=0;i<n;i++)
    {
        if(i!=n)
        {
            res=max(res,dp1[i]+dp2[i]-1);
        }
        else
        {
            res=max(res,dp1[i]);
        }
    }
    cout<<n-res<<endl;
    return 0;
}
```
