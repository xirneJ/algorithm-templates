# AcWing 7. 混合背包问题

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
struct node
{
    int v,valu;
};
node la[100010];
node a[100010];
int len1;
int len2;
int dp[1010];
int main()
{
    int n,m;
    cin>>n>>m;
    int i,j;
    for(i=0;i<n;i++)
    {
        int v,valu,s;
        cin>>v>>valu>>s;
        if(s==-1)
        {
            la[len1]={v,valu};
            len1++;
        }
        else if(s==0)
        {
            a[len2]={v,valu};
            len2++;
        }
        else
        {
            for(j=1;j<=s;j*=2)
            {
                la[len1]={v*j,valu*j};
                len1++;
                s-=j;
            }
            if(s>0)
            {
                la[len1]={v*s,valu*s};
                len1++;
            }
        }
    }
    for(i=0;i<len1;i++)
    {
        for(j=m;j>=la[i].v;j--)
        {
            dp[j]=max(dp[j],dp[j-la[i].v]+la[i].valu);
        }
    }
    for(i=0;i<len2;i++)
    {
        for(j=a[i].v;j<=m;j++)
        {
            dp[j]=max(dp[j],dp[j-a[i].v]+a[i].valu);
        }
    }
    cout<<dp[m]<<endl;
    return 0;
}
```
