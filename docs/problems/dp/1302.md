# AcWing 1072. 树的最长路径

## 模板

```cpp
#include <iostream>
#include <cstring>
#include <algorithm>
using namespace std;
int ne[20010];
int np[20010];
int valus[20010];
int h[10010];
int idx;
void add(int a,int b,int c)
{
    np[idx]=b;
    ne[idx]=h[a];
    valus[idx]=c;
    h[a]=idx++;
}
int n;
int ans=0;
int dps(int u,int fa)
{
    int fmax=0;
    int smax=0;
    int i;
    for(i=h[u];i!=-1;i=ne[i])
    {
        int r=np[i];
        if(r==fa)continue;
        int x=dps(r,u)+valus[i];
        if(x>fmax)
        {
            smax=fmax;
            fmax=x;
        }
        else if(x>smax)
        {
            smax=x;
        }
    }
    ans=max(ans,fmax+smax);
    return fmax;
}
int main()
{
    memset(h,-1,sizeof(h));
    cin>>n;
    int i,j;
    for(i=1;i<n;i++)
    {
        int l,r,c;
        cin>>l>>r>>c;
        add(l,r,c);
        add(r,l,c);
    }
    int res=dps(1,0);
    cout<<ans<<endl;
    return 0;
}
```
