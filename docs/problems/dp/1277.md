# AcWing 8. 二维费用的背包问题

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
typedef long long ll;
struct node
{
    int v,w,valu;
}arr[1010];
int dp[110][110];
int main()
{
    int n,V,W;
    cin>>n>>V>>W;
    int i,j,k;
    for(i=0;i<n;i++)
    {
        cin>>arr[i].v>>arr[i].w>>arr[i].valu;
    }
    for(i=0;i<n;i++)
    {
        for(j=V;j>=arr[i].v;j--)
        {
            for(k=W;k>=arr[i].w;k--)
            {
                dp[j][k]=max(dp[j][k],dp[j-arr[i].v][k-arr[i].w]+arr[i].valu);
            }
        }
    }
    cout<<dp[V][W]<<endl;
    return 0;
}
```
