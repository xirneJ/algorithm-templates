# AcWing 1017. 怪盗基德的滑翔翼

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
int main()
{
    int t;
    cin>>t;
    int i,j;
    while(t--)
    {
        int n;
        int res1=1;
        int res2=1;
        int arr[110]={0};
        int dp1[110]={0};
        int dp2[110]={0};
        cin>>n;
        for(i=0;i<n;i++)
        {
            cin>>arr[i];
            dp1[i]=1;
            dp2[i]=1;
        }
        for(i=0;i<n;i++)
        {
            for(j=0;j<i;j++)
            {
                if(arr[j]>arr[i])
                {
                    dp1[i]=max(dp1[i],dp1[j]+1);
                    res1=max(res1,dp1[i]);
                }
                if(arr[j]<arr[i])
                {
                    dp2[i]=max(dp2[i],dp2[j]+1);
                    res2=max(res2,dp2[i]);
                }
            }
        }

        cout<<max(res1,res2)<<endl;
    }
    return 0;
}
```
