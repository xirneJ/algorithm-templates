# AcWing 323. 战略游戏

## 模板

```cpp
#include <iostream>
#include <cstring>
#include <cstdio>
#include <algorithm>
using namespace std;
#define N 1510
int ne[N];
int np[N];
int h[N];
int dp[N][2];
int idx;
void add(int a,int b)
{
    np[idx]=b;
    ne[idx]=h[a];
    h[a]=idx++;
}
int n;
void dps(int u)
{
    int i;
    dp[u][1]=1;
    dp[u][0]=0;
    for(i=h[u];i!=-1;i=ne[i])
    {
        int r=np[i];
        dps(r);
        dp[u][1]+=min(dp[r][0],dp[r][1]);
        dp[u][0]+=dp[r][1];
    }
}
int main()
{
    int i;
    while(cin>>n)
    {
        bool st[N]={0};
        memset(h,-1,sizeof(h));
        idx=0;
        for (i=0;i<n;i++)
        {
            int l,cnt;
            scanf("%d:(%d)",&l,&cnt);
            while(cnt--)
            {
                int r;
                cin>>r;
                add(l,r);
                st[r]=true;
            }
        }
        int root=0;
        while(st[root])root++;
        dps(root);
        printf("%d\n",min(dp[root][1],dp[root][0]));
    }
    return 0;
}
```
