# AcWing 292. 炮兵阵地

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
int w[110];
vector<int>st1;
vector<int>st2[1100];
int dp[110][110][110];
int n,m;
bool check(int x)
{
    int i;
    for(i=0;i<m;i++)
    {
        if(((x>>i&1)&&(x>>i+1&1))||((x>>i&1)&&(x>>i+2&1)))
        {
            return false;
        }
    }
    return true;
}
int get(int x)
{
    int res=0;
    while(x)
    {
        if(x&1)
        {
            res++;
        }
        x>>=1;
    }
    return res;
}
void print(int x)
{
    int i;
    for(i=0;i<m;i++)
    {
        if(x>>i&1)
        {
            cout<<'1';
        }
        else
        {
            cout<<'0';
        }
    }
    cout<<' ';
}
int main()
{

    cin>>n>>m;
    int i,j,k,s;
    for(i=0;i<n;i++)
    {
        int s=0;
        for(j=0;j<m;j++)
        {
            char ch;
            cin>>ch;
            if(ch=='H')
            {
                s+=1<<j;
            }
        }
        w[i]=s;
    }

    for(i=0;i<(1<<m);i++)
    {
        if(check(i))
        {
            st1.push_back(i);
        }
    }
    int t1,t2;
    for(i=0;i<st1.size();i++)
    {
        for(j=0;j<st1.size();j++)
        {
            int a=st1[i];
            int b=st1[j];
            if(!(a&b))
            {
                st2[i].push_back(j);
            }
        }
    }

    for(i=0;i<st1.size();i++)
    {
        if(!(st1[i]&w[0]))
        {
            dp[0][i][st1.size()]=max(dp[0][i][st1.size()],get(st1[i]));
            t1=dp[0][i][st1.size()];
        }
    }

    for(i=0;i<st1.size();i++)
    {
        if(!(st1[i]&w[1]))
        {
            for(int j:st2[i])
            {
                if(!(st1[j]&w[0]))
                {
                    dp[1][i][j]=max(dp[1][i][j],dp[0][j][st1.size()]+get(st1[i]));
                }
            }
        }
    }

    //当行是j
    //前1行是k
    //前前行是s
    for(i=2;i<n;i++)
    {
        int t3=0;
        for(j=0;j<st1.size();j++)
        {
            if(!(st1[j]&w[i]))
            {
                for(int k:st2[j])
                {
                    if(!(st1[k]&w[i-1]))
                    {
                        for(int s:st2[k])
                        {
                            if(!(st1[s]&w[i-2])&&!(st1[s]&st1[j]))
                            {
                                dp[i][j][k]=max(dp[i][j][k],dp[i-1][k][s]+get(st1[j]));
                            }
                        }
                    }
                }
            }
        }
    }

    int res=0;
    if(n==1)
    {
        for(i=0;i<st1.size();i++)
        {
            res=max(res,dp[0][i][st1.size()]);
        }
    }
    else
    {
        for(i=0;i<st1.size();i++)
        {
            if(!(w[n-1]&st1[i]))
            {
                for(int j:st2[i])
                {
                    if(!(w[n-2]&st1[j]))
                    {
                        res=max(res,dp[n-1][i][j]);
                    }
                }
            }
        }
    }
    cout<<res<<endl;
    return 0;
}
```
