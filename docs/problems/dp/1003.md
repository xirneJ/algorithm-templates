# AcWing 895. 最长上升子序列

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#define N 1010
int arr[N]={0};
int dp[N]={0};
int max(int a,int b)
{
    return a>b?a:b;
}
int main()
{
    int n=0;
    scanf("%d",&n);
    int i=0;
    int j=0;
    for(i=0;i<n;i++)
    {
        scanf("%d",&arr[i]);
    }
    int res=1;
    for(i=0;i<n;i++)
    {
        dp[i]=1;
        for(j=0;j<i;j++)
        {
            if(arr[j]<arr[i])
            {
                dp[i]=max(dp[i],dp[j]+1);
                res=max(res,dp[i]);
            }
        }
    }
    printf("%d",res);
    return 0;
}
```
