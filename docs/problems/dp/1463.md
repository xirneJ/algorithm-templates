# AcWing 1091. 理想的正方形

## 模板

```cpp
#include <iostream>
#include <cstdio>
#include <algorithm>
using namespace std;
#define N 1010
int row_max[N][N];
int row_min[N][N];
int arr[N][N];
int q[N];
int n,m,k;
void get_min(int*a,int*b,int len)
{
    int left=0;
    int right=0;
    int i;
    for(i=1;i<=len;i++)
    {
        while(q[left]<=i-k&&left<=right)left++;
        while(left<=right&&a[i]<=a[q[right]])right--;
        q[++right]=i;
        b[i]=a[q[left]];
    }
}
void get_max(int*a,int*b,int len)
{
    int left=0;
    int right=0;
    int i;
    for(i=1;i<=len;i++)
    {
        while(q[left]<=i-k&&left<=right)left++;
        while(left<=right&&a[i]>=a[q[right]])right--;
        q[++right]=i;
        b[i]=a[q[left]];
    }
}
int main()
{
    cin>>n>>m>>k;
    int i,j;
    for(i=1;i<=n;i++)
    {
        for(j=1;j<=m;j++)
        {
            scanf("%d",&arr[i][j]);
        }
    }
    for(i=1;i<=n;i++)
    {
        get_min(arr[i],row_min[i],m);
        get_max(arr[i],row_max[i],m);
    }
    int a[N]={0};
    int b[N]={0};
    int c[N]={0};
    int ans=1e9;
    for(j=k;j<=m;j++)
    {
        for(i=1;i<=n;i++)
        {
            a[i]=row_min[i][j];
        }
        get_min(a,b,n);
        for(i=1;i<=n;i++)
        {
            a[i]=row_max[i][j];
        }
        get_max(a,c,n);
        for(i=k;i<=n;i++)
        {
            ans=min(ans,c[i]-b[i]);
        }
    }
    cout<<ans<<endl;
    return 0;
}
```
