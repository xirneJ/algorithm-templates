# AcWing 1010. 拦截导弹

## 模板

```cpp
#include <iostream>
#include <algorithm>
#include <cstring>
using namespace std;
int arr[1010]={0};
int dp1[1010]={0};
int dp2[1010]={0};
int n=0;
int main()
{
    int i,j;
    for(i=0;scanf("%d",&arr[i])==1;i++,n++);
    for(i=0;i<n;i++)
    {
        dp1[i]=1;
        dp2[i]=1;
    }
    int res1=1;
    int res2=1;
    for(i=0;i<n;i++)
    {
        for(j=0;j<i;j++)
        {
            if(arr[j]>=arr[i])//导弹高度不高于前一个导弹，可以相等
            {
                dp1[i]=max(dp1[i],dp1[j]+1);
                res1=max(res1,dp1[i]);
            }
            if(arr[j]<arr[i])//高度相等的可以在一个序列里面，所以高度相等的导弹不各在两个序列里面
            {
                dp2[i]=max(dp2[i],dp2[j]+1);
                res2=max(res2,dp2[i]);
            }
        }
    }
    cout<<res1<<endl;
    cout<<res2<<endl;
    return 0;
}
```
