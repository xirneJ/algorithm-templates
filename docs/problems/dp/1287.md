# AcWing 1049. 大盗阿福

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
int arr[100010];
int dp[2][100010];
int main()
{
    int t;
    cin>>t;
    while(t--)
    {
        int n,i;
        cin>>n;
        for(i=0;i<n;i++)
        {
            cin>>arr[i];
        }
        dp[1][0]=arr[0];
        for(i=1;i<n;i++)
        {
            dp[0][i]=max(dp[1][i-1],dp[0][i-1]);
            dp[1][i]=dp[0][i-1]+arr[i];
        }
        cout<<max(dp[0][n-1],dp[1][n-1])<<endl;
    }
    return 0;
}
```
