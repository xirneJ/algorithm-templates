# AcWing 423. 采药

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
struct node
{
    int t,v;
}arr[1010];
int dp[1010];
int main()
{
    int n,T;
    cin>>T>>n;
    int i,j;
    for(i=0;i<n;i++)
    {
        cin>>arr[i].t>>arr[i].v;
    }
    for(i=0;i<n;i++)
    {
        for(j=T;j>=arr[i].t;j--)
        {
            dp[j]=max(dp[j],dp[j-arr[i].t]+arr[i].v);
        }
    }
    cout<<dp[T]<<endl;
    return 0;
}
```
