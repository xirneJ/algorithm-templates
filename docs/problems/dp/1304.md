# AcWing 1075. 数字转换

## 模板

```cpp
#include <iostream>
#include <cstring>
#include <algorithm>
using namespace std;
#define N 50010
int ne[N*2];
int np[N*2];
int h[N];
int idx;
int st[N];
int sum[N];
void add(int a,int b)
{
    np[idx]=b;
    ne[idx]=h[a];
    h[a]=idx++;
}
int ans=0;
int dps(int u)
{
    int i;
    int d1=0;
    int d2=0;
    int dis=0;
    for(i=h[u];i!=-1;i=ne[i])
    {
        int r=np[i];
        int x=dps(r);
        dis=max(x,dis);
        if(x>=d1)
        {
            d2=d1;
            d1=x;
        }
        else
        {
            d2=max(d2,x);
        }
    }
    ans=max(ans,d1+d2);
    //cout<<u<<' '<<d1<<' '<<d1<<endl;
    return dis+1;
}
int main()
{
    memset(h,-1,sizeof(h));
    int n;
    cin>>n;
    int i,j;
    for(i=1;i<=n;i++)
    {
        for(j=2;j<=n/i;j++)
        {
            sum[i*j]+=i;
        }
    }
    for(i=2;i<=n;i++)
    {
        if(sum[i]<i)
        {
            st[i]=1;
            add(sum[i],i);
        }
    }
    for(i=1;i<=n;i++)
    {
        if(!st[i])
        int res=dps(i);
    }
    cout<<ans<<endl;
    return 0;
}
```
