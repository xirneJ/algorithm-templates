# AcWing 900. 整数划分

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#define N 1010
#define MAX 0x3f3f3f3f
typedef long long ll;
#define NU 1000000007
ll dp[N]={0};
ll max(ll a,ll b)
{
    return a>b?a:b;
}
int main()
{
    int n=0;
    scanf("%d",&n);
    ll i=0;
    ll j=0;
    dp[0]=1;
    for(i=1;i<=n;i++)
    {
        for(j=0;j<=n;j++)
        {
            if(j-i>=0)
            {
                dp[j]=(dp[j]+dp[j-i])%NU;
            }
        }
    }
    printf("%d\n",dp[n]);
    return 0;
}
```
