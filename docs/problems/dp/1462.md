# AcWing 1090. 绿色通道

## 模板

```cpp
#include <iostream>
#include <cstring>
#include <cstdio>
#include <algorithm>
using namespace std;
#define N 50010
int arr[N];
int f[N];
int q[N];
int n,t;
int check(int m)
{
    int left=0;
    int right=0;
    int i;
    f[0]=0;
    for(i=1;i<=n;i++)
    {
        while(q[left]<i-m-1)left++;
        f[i]=f[q[left]]+arr[i];
        while(left<=right&&f[q[right]]>=f[i])right--;
        q[++right]=i;
    }
    for(i=n-m;i<=n;i++)
    {
        if(f[i]<=t)
        {
            return 1;
        }
    }
    return 0;
}
int main()
{

    cin>>n>>t;
    int i;
    for(i=1;i<=n;i++)
    {
        scanf("%d",&arr[i]);
    }
    int left=0;
    int right=n;
    while(left<right)
    {
        int mid=left+right>>1;
        if(check(mid))
        {
            right=mid;
        }
        else
        {
            left=mid+1;
        }
    }
    cout<<right<<endl;
    return 0;
}
```
