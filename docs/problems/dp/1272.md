# AcWing 1021. 货币系统

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
typedef long long ll;
int arr[20];
ll dp[3010];
int main()
{
    int n,m;
    cin>>n>>m;
    int i,j;
    for(i=0;i<n;i++)
    {
        cin>>arr[i];
    }
    dp[0]=1;
    for(i=0;i<n;i++)
    {
        for(j=arr[i];j<=m;j++)
        {
            dp[j]+=dp[j-arr[i]];
        }
    }
    cout<<dp[m]<<endl;
    return 0;
}
```
