# AcWing 1018. 最低通行费

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
int g[110][110];
int dp[110][110];
int main()
{
    int n;
    cin>>n;
    int i,j;
    for(i=0;i<n;i++)
    {
        for(j=0;j<n;j++)
        {
            cin>>g[i][j];
        }
    }
    dp[0][0]=g[0][0];
    for(i=1;i<n;i++)
    {
        dp[i][0]=dp[i-1][0]+g[i][0];
        dp[0][i]=dp[0][i-1]+g[0][i];
    }
    for(i=1;i<n;i++)
    {
        for(j=1;j<n;j++)
        {
            dp[i][j]=min(dp[i-1][j],dp[i][j-1])+g[i][j];
        }
    }
    cout<<dp[n-1][n-1]<<endl;
    return 0;
}
```
