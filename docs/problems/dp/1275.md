# AcWing 1019. 庆功会

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
struct node
{
    int cost,valu;
}arr[5010];
int len=0;
int dp[6010];
int main()
{
    int n,m;
    cin>>n>>m;
    int i,j;
    for(i=0;i<n;i++)
    {
        int cost,valu,c;
        cin>>cost>>valu>>c;
        for(j=1;j<=c;j*=2)
        {
            arr[len].cost=cost*j;
            arr[len].valu=valu*j;
            len++;
            c-=j;
        }
        if(c>0)
        {
            arr[len].cost=c*cost;
            arr[len].valu=c*valu;
            len++;
        }
    }
    for(i=0;i<len;i++)
    {
        for(j=m;j>=arr[i].cost;j--)
        {
            dp[j]=max(dp[j],dp[j-arr[i].cost]+arr[i].valu);
        }
    }
    cout<<dp[m]<<endl;
    return 0;
}
```
