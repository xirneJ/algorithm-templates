# AcWing 5. 多重背包问题 II

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
#define N 100010
struct l
{
    int v;
    int valu;
    int c;
}arr1[N];
struct r
{
    int v;
    int valu;
}arr2[N];
int dp[N];
int main()
{
    int n,V;
    cin>>n>>V;
    int i,j;
    int len=0;
    for(i=0;i<n;i++)
    {
        cin>>arr1[i].v;
        cin>>arr1[i].valu;
        cin>>arr1[i].c;
    }
    for(i=0;i<n;i++)
    {
        int v=arr1[i].v;
        int valu=arr1[i].valu;
        int c=arr1[i].c;
        for(j=1;j<=c;j*=2)
        {
            arr2[len].v=v*j;
            arr2[len].valu=valu*j;
            c-=j;
            len++;
        }
        if(c>0)
        {
            arr2[len].v=v*c;
            arr2[len].valu=valu*c;
            len++;
        }
    }
    for(i=0;i<len;i++)
    {
        for(j=V;j>=arr2[i].v;j--)
        {
            dp[j]=max(dp[j],dp[j-arr2[i].v]+arr2[i].valu);
        }
    }
    cout<<dp[V]<<endl;
    return 0;
}
```
