# AcWing 1084. 数字游戏 II

## 模板

```cpp
#include <iostream>
#include <vector>
#include <cstring>
#include <cstdio>
#include <algorithm>
using namespace std;
#define N 20
int f[N][N][210];
int a,b,p;
int dp(int n)
{
    if(!n)return (n+1)%p?0:1;
    vector<int>s;
    while(n)
    {
        s.push_back(n%10);
        n/=10;
    }
    n=s.size();
    int i,j;
    int res=0;
    for(i=1;i<n;i++)
    {
        for(j=1;j<10;j++)
        {
            res+=f[i][j][0];
        }
    }
    int sum=0;
    for(i=n-1;i>=0;i--)
    {
        int x=s[i];
        if(i==n-1)
        {
            for(j=1;j<x;j++)
            {
                res+=f[i+1][j][(p-sum%p)%p];
            }
        }
        else
        {
            for(j=0;j<x;j++)
            {
                res+=f[i+1][j][(p-sum%p)%p];
            }
        }
        sum+=x;
        if((!i)&&sum%p==0)res++;
    }
    return res;
}
int main()
{
    while(scanf("%d %d %d",&a,&b,&p)==3)
    {
        memset(f,0,sizeof(f));
        if(a>b)swap(a,b);
        int i,j,k,t;
        for(i=0;i<10;i++)
        {
            f[1][i][i%p]++;
        }
        for(k=2;k<N;k++)
        {
            for(i=0;i<10;i++)
            {
                for(j=0;j<10;j++)
                {
                    for(t=0;t<p;t++)
                    {
                        f[k][i][(i+t)%p]+=f[k-1][j][t];
                    }
                }
            }
        }
        cout<<dp(b)-dp(a-1)<<endl;
    }
    return 0;
}
```
