# AcWing 10. 有依赖的背包问题

## 模板

```cpp
#include <iostream>
#include <algorithm>
#include <cstring>
using namespace std;
struct node
{
    int v,valu;
}arr[110];
int n,m;
int ne[110];
int np[110];
int h[110];
int idx=0;
int dp[110][110];
void add(int a,int b)
{
    np[idx]=b;
    ne[idx]=h[a];
    h[a]=idx++;
}
void dps(int u)
{
    int i,j,k;
    for(i=h[u];i!=-1;i=ne[i])
    {
        int son=np[i];
        dps(son);
        for(j=m-arr[u].v;j>=0;j--)
        {
            for(k=0;k<=j;k++)
            {
                dp[u][j]=max(dp[u][j],dp[u][j-k]+dp[son][k]);
            }
        }
    }
    for(i=m;i>=arr[u].v;i--)
    {
        dp[u][i]=dp[u][i-arr[u].v]+arr[u].valu;
    }
    for(i=0;i<arr[u].v;i++)
    {
        dp[u][i]=0;
    }
}
int main()
{
    memset(h,-1,sizeof(h));
    cin>>n>>m;
    int i;
    int fa=0;
    for(i=1;i<=n;i++)
    {
        int p;
        cin>>arr[i].v>>arr[i].valu>>p;
        if(p==-1)
        {
            fa=i;
        }
        else
        {
            add(p,i);
        }
    }
    dps(fa);
    cout<<dp[fa][m]<<endl;
    return 0;
}
```
