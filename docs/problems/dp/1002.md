# AcWing 898. 数字三角形

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
int dp[510][510];
int main()
{
    int n;
    cin>>n;
    int arr[510][510]={0};
    int i,j;
    for(i=1;i<=n;i++)
    {
        for(j=1;j<=i;j++)
        {
            cin>>arr[i][j];
        }
    }
    for(j=1;j<=n;j++)
    {
        dp[n][j]=arr[n][j];
    }
    for(i=n-1;i>=1;i--)
    {
        for(j=1;j<=i;j++)
        {
            dp[i][j]=max(dp[i+1][j],dp[i+1][j+1])+arr[i][j];
        }
    }
    cout<<dp[1][1]<<endl;
    return 0;
}
```
