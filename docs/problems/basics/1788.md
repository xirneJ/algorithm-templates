# AcWing 99. 激光炸弹

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
int sum[5010][5010];
int max_x=0;
int max_y=0;
int sum1=0;
int main()
{
    int n,r;
    cin>>n>>r;
    int i,j;
    for(i=0;i<n;i++)
    {
        int x,y,v;
        cin>>x>>y>>v;
        sum[x][y]+=v;
        sum1+=v;
    }
    if(r>=5001)
    {
        cout<<sum1<<endl;
        return 0;
    }
    for(i=1;i<=5000;i++)
    {
        sum[0][i]=sum[0][i]+sum[0][i-1];
        sum[i][0]=sum[i][0]+sum[i-1][0];
    }
    for(i=1;i<=5000;i++)
    {
        for(j=1;j<=5000;j++)
        {
            sum[i][j]=sum[i][j]+sum[i-1][j]+sum[i][j-1]-sum[i-1][j-1];
        }
    }
    int ans=0;
    for(i=0;i<=5000;i++)
    {
        for(j=0;j<=5000;j++)
        {
            int res=sum[i][j];
            if(i>=r)
            {
                res-=sum[i-r][j];
            }
            if(j>=r)
            {
                res-=sum[i][j-r];
            }
            if(i>=r&&j>=r)
            {
                res+=sum[i-r][j-r];
            }
            ans=max(res,ans);
        }
    }
    cout<<ans<<endl;
    return 0;
}
```
