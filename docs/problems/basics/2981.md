# AcWing 2816. 判断子序列

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#define N 100010
int arr1[N]={0};
int arr2[N]={0};
int main()
{
    int len1,len2;
    int i=0;
    int j=0;
    scanf("%d %d",&len1,&len2);
    for(i=0;i<len1;i++)
    {
        scanf("%d",&arr1[i]);
    }
    for(i=0;i<len2;i++)
    {
        scanf("%d",&arr2[i]);
    }
    i=0;
    j=0;
    while(i<len1&&j<len2)
    {
        if(arr1[i]==arr2[j])
        {
            i++;
        }
        j++;
    }
    if(i==len1)
    {
        printf("Yes\n");
    }
    else
    {
        printf("No\n");
    }
    return 0;
}
```
