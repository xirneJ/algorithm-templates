# AcWing 800. 数组元素的目标和

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#define N 100010
int arr1[N]={0};
int arr2[N]={0};
int main()
{
    int len1,len2,aim;
    scanf("%d %d %d",&len1,&len2,&aim);
    int i=0;
    int j=0;
    for(i=0;i<len1;i++)
    {
        scanf("%d",&arr1[i]);
    }
    for(i=0;i<len2;i++)
    {
        scanf("%d",&arr2[i]);
    }
    for(i=0,j=len2-1;i<len1,j>=0;i++)
    {
        while(j>=0&&(arr1[i]+arr2[j])>aim)
        {
            j--;
        }
        if((arr1[i]+arr2[j])==aim)
        {
            printf("%d %d",i,j);
            break;
        }
    }
    return 0;
}
```
