# AcWing 799. 最长连续不重复子序列

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#define N 100010
int arr[N]={0};
int count[N]={0};
int max(int a,int b)
{
    return a>b?a:b;
}
int main()
{
    int n=0;
    int i=0;
    int j=0;
    scanf("%d",&n);
    for(i=0;i<n;i++)
    {
        scanf("%d",&arr[i]);
    }
    int res=-1;
    for(i=0,j=0;i<n;i++)
    {
        count[arr[i]]++;
        while(count[arr[i]]>1)
        {
            count[arr[j]]--;
            j++;
        }
        res=max(res,i-j+1);
    }
    printf("%d",res);
    return 0;
}
```
