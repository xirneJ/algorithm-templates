# AcWing 794. 高精度除法

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#define N 100010
int len1=0;
int len2=0;
void reverse(int*c,int len)//反转函数，将数组里面的内容反转过来
{
    int i=0;
    for(i=0;i<len/2;i++)
    {
        int num=c[i];
        c[i]=c[len-1-i];
        c[len-1-i]=num;
    }
}

void mdiv(int*a,int*c,int b,int*res)//除法的主体函数
{
    int i=0;
    int t=0;
    for(i=len1-1;i>=0;i--)
    {
        t=t*10+a[i];
        c[len2++]=t/b;
        t%=b;
    }

    (*res)=t;//记录余数
    reverse(c,len2);
    while(len2>1&&c[len2-1]==0)//去除前导零
    {
        len2--;
    }
}

int main()
{
    char a[N]={0};
    int b=0;
    int A[N]={0};
    int C[N]={0};
    scanf("%s %d",a,&b);

    int i=0;
    len1=strlen(a);
    int res=0;

    for(i=0;i<len1;i++)//将字符转化为数字
    {
        A[i]=a[len1-1-i]-'0';
    }

    mdiv(A,C,b,&res);
    for(i=len2-1;i>=0;i--)//打印结果
    {
        printf("%d",C[i]);
    }
    printf("\n");
    printf("%d",res);
    return 0;
}
```
