# AcWing 100. 增减序列

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
typedef long long ll;
ll arr1[100010];
ll arr2[100010];
//本题差分后只有三种炒作
//b1+1....b(2-n)-1或b1-1...b(2-n)-1
//b(2-n)+1...b(n+1)-1或b(2-n)-1...b(n+1)-1;
//b(2-n)+1...b(2-n)-1或b(2-n)+1..b(2-n)-1
int main()
{
    int n;
    cin>>n;
    int i;
    for(i=0;i<n;i++)
    {
        cin>>arr1[i];
    }
    arr2[0]=arr1[0];
    for(i=1;i<n;i++)
    {
        arr2[i]=arr1[i]-arr1[i-1];
    }
    ll p=0;//记录大于0的值
    ll q=0;//记录小于0的值
    for(i=1;i<n;i++)
    {
        if(arr2[i]>0)
        {
            p+=arr2[i];//由于每次只能加以或减一，p要加上arr2[i]，而不是+1
        }
        else if(arr2[i]<0)
        {
            q+=arr2[i];
        }
    }
    cout<<max(abs(q),p)<<endl;
    cout<<abs(p-abs(q))+1<<endl;//有|p-q|+1种总共
    return 0;
}
```
