# AcWing 90. 64位整数乘法

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
typedef long long ll;
ll qmi(ll a,ll b,ll p)
{
    ll res=0;
    ll num=a;
    while(b)
    {
        if(b&1)
        {
            res=(res+num)%p;
        }
        num=num*2%p;
        b/=2;
    }
    return res;
}
int main()
{
    ll a,b,p;
    cin>>a>>b>>p;
    cout<<qmi(a,b,p)<<endl;
    return 0;
}
```
