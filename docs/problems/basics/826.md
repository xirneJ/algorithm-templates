# AcWing 792. 高精度减法

## 模板

```cpp
#include <stdio.h>
#include <string.h>
#define N 100010
int len1=0;//a(A)数组的长度
int len2=0;//b(B)数组的长度
int len3=0;//C数组的长度

int C[N]={0};

int cmp(char* a,char* b)//判断函数，判断a和b的大小
{
    int i=0;
    if(len1>len2)
    {
        return 1;
    }
    else if(len1<len2)
    {
        return 0;
    }
    else
    {
        for(i=0;i<len1;i++)
        {
            if(a[i]!=b[i])
            {
                return a[i]>b[i];
            }
        }
        return 1;
    }
}

void sub(int*A,int*B)//高精度减法的主体函数
{
    int t=0;
    int i=0;
    for(i=0;i<len1;i++)
    {
        t=A[i]-t;
        if(i<len2)
        {
            t-=B[i];
        }
        C[len3++]=(t+10)%10;
        if(t>=0)
        {
            t=0;
        }
        else
        {
            t=1;
        }
    }
    while(C[len3-1]==0&&len3>1)//去除前导0
    {
        len3--;
    }
}
int main()
{
    char a[N]={0};
    char b[N]={0};
    int A[N]={0};
    int B[N]={0};

    scanf("%s %s",a,b);
    len1=strlen(a);
    len2=strlen(b);

    int i=0;
    for(i=0;i<len1;i++)//转化a字符串
    {
        A[i]=a[len1-1-i]-'0';
    }
    for(i=0;i<len2;i++)//转化b字符串
    {
        B[i]=b[len2-1-i]-'0';
    }

    if(cmp(a,b))//如果a>=b就执行a-b
    {
        sub(A,B);
        for(i=len3-1;i>=0;i--)
        {
            printf("%d",C[i]);
        }
    }

    else//如果a<b就执行b-a,最后在结果上加个'-'
    {
        int num=len1;//两个减数的位置变化了，所以len1和len2的量要交换一下
        len1=len2;
        len2=num;

        sub(B,A);
        printf("-");
        for(i=len3-1;i>=0;i--)
        {
            printf("%d",C[i]);
        }
    }
    return 0;
}
```
