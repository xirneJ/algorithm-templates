# AcWing 798. 差分矩阵

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#define N 1010
int arr[N][N]={0};
int C[N][N]={0};
int sum[N][N]={0};
int main()
{
    int l,r,q;
    scanf("%d %d %d",&l,&r,&q);
    int i=0;
    int j=0;
    for(i=0;i<l;i++)
    {
        for(j=0;j<r;j++)
        {
            scanf("%d",&arr[i][j]);
        }
    }
    C[0][0]=arr[0][0];
    for(i=1;i<l;i++)
    {
        C[i][0]=arr[i][0]-arr[i-1][0];
    }
    for(j=1;j<r;j++)
    {
        C[0][j]=arr[0][j]-arr[0][j-1];
    }
    for(i=1;i<l;i++)
    {
        for(j=1;j<r;j++)
        {
            C[i][j]=arr[i][j]-arr[i-1][j]-arr[i][j-1]+arr[i-1][j-1];
        }
    }
    while(q--)
    {
        int x1,y1,x2,y2,v;
        scanf("%d %d %d %d %d",&x1,&y1,&x2,&y2,&v);
        x1--;
        x2--;
        y1--;
        y2--;
        C[x1][y1]+=v;
        if((x2+1)<l)
        {
            C[x2+1][y1]-=v;
        }
        if((y2+1)<r)
        {
            C[x1][y2+1]-=v;
        }
        if((x2+1)<l&&(y2+1)<r)
        {
            C[x2+1][y2+1]+=v;
        }
    }
    sum[0][0]=C[0][0];
    for(i=1;i<l;i++)
    {
        sum[i][0]=sum[i-1][0]+C[i][0];
    }
    for(i=1;i<r;i++)
    {
        sum[0][i]=sum[0][i-1]+C[0][i];
    }
    for(i=1;i<l;i++)
    {
        for(j=1;j<r;j++)
        {
            sum[i][j]=sum[i-1][j]+sum[i][j-1]-sum[i-1][j-1]+C[i][j];
        }
    }
    for(i=0;i<l;i++)
    {
        for(j=0;j<r;j++)
        {
            printf("%d ",sum[i][j]);
        }
        printf("\n");
    }
    return 0;
}
```
