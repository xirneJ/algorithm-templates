# AcWing 795. 前缀和

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#define N 100010
int main()
{
    int n=0;
    int i=0;
    int q=0;
    int arr[N]={0};
    scanf("%d %d",&n,&q);
    for(i=0;i<n;i++)
    {
        scanf("%d",&arr[i]);
    }
    int sum[N]={0};
    sum[0]=arr[0];
    for(i=1;i<n;i++)
    {
        sum[i]=sum[i-1]+arr[i];
    }
    while(q--)
    {
        int l,r;
        scanf("%d %d",&l,&r);
        l--;
        r--;
        int res=sum[r];
        if(l>=1)
        {
            res-=sum[l-1];
        }
        printf("%d\n",res);
    }
    return 0;
}
```
