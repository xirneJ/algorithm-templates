# AcWing 97. 约数之和

## 模板

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
int qmi(int a,int k)
{
    a=a%9901;
    int res=1;
    while(k)
    {
        if(k&1)
        {
            res=res*a%9901;
        }
        k>>=1;
        a=a*a%9901;
    }
    return res;
}
int sum(int p,int c)
{
    if(c==1)
    {
        return 1;
    }
    if(c%2==0)
    {
        return (qmi(p,c/2)+1)*sum(p,c/2)%9901;
    }
    else
    {
        return (sum(p,c-1)+qmi(p,c-1))%9901;
    }
}
int main()
{
    int a,b,x;
    cin>>a>>b;
    int res=1;
    x=a;
    int i;
    for(i=2;i<=x/i;i++)
    {
        if(x%i==0)
        {
            int s=0;
            while(x%i==0)
            {
                s++;
                x/=i;
            }
            res=res*sum(i,s*b+1)%9901;
        }
    }
    if(x>1)
    {
        res=res*sum(x,b+1)%9901;
    }
    if(a==0)
    {
        res=0;
    }
    cout<<res<<endl;
    return 0;
}
```
