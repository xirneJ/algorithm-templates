# AcWing 796. 子矩阵的和

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#define N 1010
int arr[N][N]={0};
int sum[N][N]={0};
int main()
{
    int l,r,q;
    scanf("%d %d %d",&l,&r,&q);
    int i=0;
    int j=0;
    for(i=0;i<l;i++)
    {
        for(j=0;j<r;j++)
        {
            scanf("%d",&arr[i][j]);
        }
    }
    sum[0][0]=arr[0][0];
    for(i=1;i<l;i++)
    {
        sum[i][0]=sum[i-1][0]+arr[i][0];
    }
    for(j=1;j<r;j++)
    {
        sum[0][j]=sum[0][j-1]+arr[0][j];
    }
    for(i=1;i<l;i++)
    {
        for(j=1;j<r;j++)
        {
            sum[i][j]=sum[i-1][j]+sum[i][j-1]-sum[i-1][j-1]+arr[i][j];
        }
    }
    while(q--)
    {
        int x1,y1,x2,y2;
        scanf("%d %d %d %d",&x1,&y1,&x2,&y2);
        x1--;
        y1--;
        x2--;
        y2--;
        int res=sum[x2][y2];
        if(y1>=1)
        {
            res-=sum[x2][y1-1];
        }
        if(x1>=1)
        {
            res-=sum[x1-1][y2];
        }
        if(x1>=1&&y1>=1)
        {
            res+=sum[x1-1][y1-1];
        }
        printf("%d\n",res);
    }
    return 0;
}
```
