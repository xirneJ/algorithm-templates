# AcWing 793. 高精度乘法

## 模板

```cpp
#include <stdio.h>
#include <string.h>
#define N 100010
int len1=0;
int len2=0;
int C[N]={0};
void mul(int*a,int b)//高精度乘法的主体函数
{
    int i=0;
    int t=0;
    for(i=0;i<len1||t!=0;i++)
    {
        if(i<len1)
        {
            t=t+a[i]*b;
        }
        C[len2++]=t%10;
        t/=10;
    }
}

int main()
{
    char a[N]={0};
    int A[N]={0};
    int b=0;
    int i=0;
    scanf("%s %d",a,&b);//输入数据

    if(b==0)//检测一下，防止结果有一串0，如12345*0结果为00000
    {
        printf("0");
        return 0;
    }

    len1=strlen(a);
    for(i=0;i<len1;i++)//数据转换
    {
        A[i]=a[len1-1-i]-'0';
    }

    mul(A,b);
    for(i=len2-1;i>=0;i--)//打印结果
    {
        printf("%d",C[i]);
    }
    return 0;
}
```
