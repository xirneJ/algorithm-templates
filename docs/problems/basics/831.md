# AcWing 797. 差分

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#define N 100010
int arr[N]={0};
int C[N]={0};
int sum[N]={0};
int main()
{
    int n,q;
    scanf("%d %d",&n,&q);
    int i=0;
    for(i=0;i<n;i++)
    {
        scanf("%d",&arr[i]);
    }
    C[0]=arr[0];
    for(i=1;i<n;i++)
    {
        C[i]=arr[i]-arr[i-1];
    }
    while(q--)
    {
        int l,r,v;
        scanf("%d %d %d",&l,&r,&v);
        l--;
        r--;
        C[l]+=v;
        if((r+1)<n)
        {
            C[r+1]-=v;
        }
    }
    sum[0]=C[0];
    for(i=1;i<n;i++)
    {
        sum[i]=sum[i-1]+C[i];
    }
    for(i=0;i<n;i++)
    {
        printf("%d ",sum[i]);
    }
    return 0;
}
```
