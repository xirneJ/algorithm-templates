# AcWing 95. 费解的开关

## 模板

```cpp
#include <algorithm>
#include <iostream>
#include <cstring>
using namespace std;
char g[6][6];
char bg[6][6];
int nx[5]={1,-1,0,0,0};
int ny[5]={0,0,1,-1,0};
void turn(int x,int y)
{
    int i;
    for(i=0;i<5;i++)
    {
        int new_x=nx[i]+x;
        int new_y=ny[i]+y;
        if(new_x>=0&&new_x<5&&new_y>=0&&new_y<5)
        {
            g[new_x][new_y]=g[new_x][new_y]^1;
        }
    }
}
int main()
{
    int t;
    cin>>t;
    while(t--)
    {
        int i,j,op;
        int res=100;
        for(i=0;i<5;i++)
        {
            cin>>bg[i];
        }
        for(op=0;op<(1<<5);op++)
        {
            memcpy(g,bg,sizeof(g));
            int cnt=0;
            for(i=0;i<5;i++)
            {
                if(op>>i&1)
                {
                    turn(0,i);
                    cnt++;
                }
            }
            for(i=0;i<4;i++)
            {
                for(j=0;j<5;j++)
                {
                    if(g[i][j]=='0')
                    {
                        turn(i+1,j);
                        cnt++;
                    }
                }
            }
            int st=1;
            for(i=0;i<5;i++)
            {
                if(g[4][i]=='0')
                {
                    st=0;
                }
            }
            if(st)
            {
                res=min(res,cnt);
            }
        }
        if(res>6)
        {
            cout<<"-1"<<endl;
        }
        else
        {
            cout<<res<<endl;
        }
    }
    return 0;
}
```
