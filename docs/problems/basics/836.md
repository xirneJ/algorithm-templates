# AcWing 802. 区间和

## 模板

```cpp
#include <stdio.h>
#include <stdlib.h>
#define N 100010
typedef struct l
{
    int p;
    int v;
}arry;
arry arr[N]={0};
int sum[N]={0};
int cmp(const void*e1,const void*e2)
{
    arry*a1=(arry*)e1;
    arry*a2=(arry*)e2;
    return a1->p-a2->p;
}
int find_left(arry*arr,int left,int right,int x)
{
    while(left<=right)
    {
        int mid=(right-left)/2+left;
        if(arr[mid].p<x)
        {
            left=mid+1;
        }
        else if(arr[mid].p>x)
        {
            right=mid-1;
        }
        else if(arr[mid].p==x)
        {
            return mid;
        }
    }
    return left;
}
int find_right(arry*arr,int left,int right,int x)
{
    while(left<=right)
    {
        int mid=(right-left)/2+left;
        if(arr[mid].p<x)
        {
            left=mid+1;
        }
        else if(arr[mid].p>x)
        {
            right=mid-1;
        }
        else if(arr[mid].p==x)
        {
            return mid;
        }
    }
    return right;
}
int main()
{
    int n,q;
    scanf("%d %d",&n,&q);
    int i=0;
    int j=-1;
    for(i=0;i<n;i++)
    {
        scanf("%d %d",&arr[i].p,&arr[i].v);
    }
    qsort(arr,n,sizeof(arry),cmp);
    for(i=0;i<n;i++)
    {
        if(j==-1||arr[i].p!=arr[j].p)
        {
            j++;
            arr[j]=arr[i];
        }
        else
        {
            arr[j].v+=arr[i].v;
        }
    }
    sum[0]=arr[0].v;
    for(i=1;i<=j;i++)
    {
        sum[i]=sum[i-1]+arr[i].v;
    }
    while(q--)
    {
        int l,r;
        scanf("%d %d",&l,&r);
        int res=sum[find_right(arr,0,j,r)];
        int p=find_left(arr,0,j,l);
        if(p>=1)
        {
            res-=sum[p-1];
        }
        printf("%d\n",res);
    }
    return 0;
}
```
